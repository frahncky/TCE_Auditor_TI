% O banco é apresentado por nível: questões do nível e, logo depois,
% suas respectivas resoluções comentadas. Não há bloco de gabarito separado.
\ExplSyntaxOn
\tl_new:N \l_tce_nivel_alvo_tl

\cs_new_protected:Npn \tce_define_nivel:n #1
  { \tl_set:Nn \l_tce_nivel_alvo_tl {#1} }

\prg_new_conditional:Npnn \tce_questao_do_nivel:n #1 {TF}
  {
    \str_case:VnF \l_tce_nivel_alvo_tl
      {
        {N1}{\tl_if_in:nnTF {#1}{N1}{\prg_return_true:}{\prg_return_false:}}
        {N2}{\tl_if_in:nnTF {#1}{N2}{\prg_return_true:}{\prg_return_false:}}
        {N3}{\tl_if_in:nnTF {#1}{N3}{\prg_return_true:}{\prg_return_false:}}
        {N4}{\tl_if_in:nnTF {#1}{N4}{\prg_return_true:}{\prg_return_false:}}
      }
      {\prg_return_false:}
  }

\prg_new_conditional:Npnn \tce_solucao_do_nivel:n #1 {TF}
  {
    \str_case:VnF \l_tce_nivel_alvo_tl
      {
        {N1}{\regex_match:nnTF {\.(0[1-9]|10)\Z} {#1}{\prg_return_true:}{\prg_return_false:}}
        {N2}{\regex_match:nnTF {\.(1[1-9]|2[0-5])\Z} {#1}{\prg_return_true:}{\prg_return_false:}}
        {N3}{\regex_match:nnTF {\.(2[6-9]|3[0-9]|40)\Z} {#1}{\prg_return_true:}{\prg_return_false:}}
        {N4}{\regex_match:nnTF {\.(4[1-9]|50)\Z} {#1}{\prg_return_true:}{\prg_return_false:}}
      }
      {\prg_return_false:}
  }

\NewDocumentCommand{\TCEDefineNivel}{m}
  {\tce_define_nivel:n{#1}}
\NewDocumentCommand{\TCEQuestaoDoNivelTF}{m +m +m}
  {\tce_questao_do_nivel:nTF{#1}{#2}{#3}}
\NewDocumentCommand{\TCESolucaoDoNivelTF}{m +m +m}
  {\tce_solucao_do_nivel:nTF{#1}{#2}{#3}}
\ExplSyntaxOff

\NewDocumentCommand{\TCERenderQuestao}{m +m}{%
  \csname tce_prepara_titulo_questao:n\endcsname{#1}%
  \begin{tcolorbox}[
    colback=white,
    colframe=azulMedio,
    title={Questão \csname tce_titulo_questao:\endcsname},
    fonttitle=\bfseries
  ]
  #2
  \csname tce_opcoes_ce:\endcsname
  \end{tcolorbox}%
}

\NewDocumentCommand{\TCERenderSolucao}{m +m}{%
  \begin{tcolorbox}[
    colback=cinzaClaro,
    colframe=verdeAcerto,
    title={Resolução #1},
    fonttitle=\bfseries
  ]
  #2
  \end{tcolorbox}%
}

\NewDocumentCommand{\TCEInputQuestoesNivel}{m m}{%
  \begingroup
  \TCEDefineNivel{#1}%
  \RenewDocumentCommand{\section}{s o m}{}%
  \RenewDocumentCommand{\subsection}{s o m}{}%
  \RenewEnviron{questao}[1]{%
    \TCEQuestaoDoNivelTF{##1}{\TCERenderQuestao{##1}{\BODY}}{}%
  }%
  \input{#2}%
  \endgroup
}

\NewDocumentCommand{\TCEInputSolucoesNivel}{m m}{%
  \begingroup
  \TCEDefineNivel{#1}%
  \RenewDocumentCommand{\section}{s o m}{}%
  \RenewDocumentCommand{\subsection}{s o m}{}%
  \RenewEnviron{solucao}[1]{%
    \TCESolucaoDoNivelTF{##1}{\TCERenderSolucao{##1}{\BODY}}{}%
  }%
  \input{#2}%
  \endgroup
}

\NewDocumentCommand{\TCENivelCompleto}{m m m m}{%
  \subsection*{\nivel{#1}\enspace #2}%
  \TCEInputQuestoesNivel{#1}{#3}%
  \subsubsection*{Resoluções comentadas \nivel{#1}}%
  \TCEInputSolucoesNivel{#1}{#4}%
}

\NewDocumentCommand{\TCEBancoCapitulo}{m m}{%
  \chapter{#1}%
  \TCENivelCompleto{N1}{Fundamento competitivo}{questoes/#2}{solucoes/#2}%
  \TCENivelCompleto{N2}{Aplicação}{questoes/#2}{solucoes/#2}%
  \TCENivelCompleto{N3}{Integração de concurso}{questoes/#2}{solucoes/#2}%
  \TCENivelCompleto{N4}{Auditoria / avançado}{questoes/#2}{solucoes/#2}%
}
