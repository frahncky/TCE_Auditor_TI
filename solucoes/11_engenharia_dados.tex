\section{Gabarito e resoluções comentadas — Engenharia de Dados}

\begin{solucao}{11.01}\textbf{CERTO.} Dependência de atributo não-chave em parte da chave composta é dependência parcial e viola 2FN.\end{solucao}
\begin{solucao}{11.02}\textbf{ERRADO.} BCNF é mais restritiva. Toda BCNF satisfaz 3FN, mas nem toda 3FN satisfaz BCNF.\end{solucao}
\begin{solucao}{11.03}\textbf{CERTO.} B-tree é versátil para igualdade/intervalo, mas o otimizador considera seletividade, estatísticas, custo e ordem das colunas.\end{solucao}
\begin{solucao}{11.04}\textbf{ERRADO.} Persistência após commit é durabilidade. Atomicidade é tudo-ou-nada.\end{solucao}
\begin{solucao}{11.05}\textbf{CERTO.} Granularidade define o significado da linha e condiciona quais medidas podem ser agregadas.\end{solucao}
\begin{solucao}{11.06}\textbf{CERTO.} No ELT, o dado é carregado antes de parte relevante das transformações.\end{solucao}
\begin{solucao}{11.07}\textbf{ERRADO.} O trade-off CAP se torna inevitável na presença de partição de rede; não é escolha permanente em qualquer estado do sistema.\end{solucao}
\begin{solucao}{11.08}\textbf{CERTO.} Lineage sustenta rastreabilidade e reprodutibilidade de indicadores e relatórios.\end{solucao}
\begin{solucao}{11.09}\textbf{ERRADO.} Chave estrangeira trata referência, não unicidade na tabela filha.\end{solucao}
\begin{solucao}{11.10}\textbf{CERTO.} Idempotência evita efeitos duplicados em reexecuções ou retries.\end{solucao}

\begin{solucao}{11.11}\textbf{B.} NomeAluno e Curso dependem apenas de AlunoID; separar Aluno remove a dependência parcial. Criar surrogate key sem alterar dependências não resolve a anomalia.\end{solucao}
\begin{solucao}{11.12}\textbf{B.} LEFT JOIN preserva órgãos sem contrato; a agregação deve tratar NULL para produzir zero.\end{solucao}
\begin{solucao}{11.13}\textbf{CERTO.} Predicado no WHERE sobre a direita elimina linhas em que os campos são NULL, neutralizando o efeito esperado do LEFT JOIN.\end{solucao}
\begin{solucao}{11.14}\textbf{A.} O índice começa por \texttt{orgao\_id} e, depois, \texttt{data\_evento}; esse padrão aproveita o prefixo do índice.\end{solucao}
\begin{solucao}{11.15}\textbf{CERTO.} Funções/conversões podem tornar o predicado não sargable, salvo suporte específico.\end{solucao}
\begin{solucao}{11.16}\textbf{B.} A segunda gravação sobrescreve a primeira sem perceber: lost update.\end{solucao}
\begin{solucao}{11.17}\textbf{B.} Valor total da nota repetido em cada item é multiplicado pela quantidade de itens.\end{solucao}
\begin{solucao}{11.18}\textbf{CERTO.} Saldo é semi-aditivo: soma entre entidades, mas não simplesmente ao longo do tempo.\end{solucao}
\begin{solucao}{11.19}\textbf{B.} CDC permite processar alterações, reduzindo volume; reprocessamento, exclusões e reconciliação continuam necessários.\end{solucao}
\begin{solucao}{11.20}\textbf{CERTO.} Completo não significa correto. Um campo preenchido com valor errado mantém completude e perde acurácia.\end{solucao}
\begin{solucao}{11.21}\textbf{A.} Identificadores precisam de normalização, validação e regras rastreáveis de deduplicação.\end{solucao}
\begin{solucao}{11.22}\textbf{CERTO.} Catálogo organiza conhecimento, mas não executa controles de qualidade, acesso ou continuidade.\end{solucao}
\begin{solucao}{11.23}\textbf{B.} Retry é normal em sistemas resilientes; idempotência e identificadores estáveis controlam duplicação.\end{solucao}
\begin{solucao}{11.24}\textbf{CERTO.} Em séries temporais, misturar futuro e passado na divisão pode gerar avaliação irrealista.\end{solucao}
\begin{solucao}{11.25}\textbf{B.} PITR precisa ser demonstrado por restauração real com backup base e logs; a mera existência do WAL é insuficiente.\end{solucao}

\begin{solucao}{11.26}\textbf{D.} I, II e III estão corretas. BCNF é mais forte; desnormalização pode ser consciente em analytics.\end{solucao}
\begin{solucao}{11.27}\textbf{CERTO.} MVCC reduz bloqueios de leitura, mas disputas de escrita ainda podem produzir deadlocks.\end{solucao}
\begin{solucao}{11.28}\textbf{A.} Mudança de distribuição pode tornar estimativas de cardinalidade erradas se estatísticas estiverem defasadas.\end{solucao}
\begin{solucao}{11.29}\textbf{B.} SCD2 preserva versões históricas e período de validade.\end{solucao}
\begin{solucao}{11.30}\textbf{CERTO.} Em partição, maximizar disponibilidade pode exigir aceitar consistência mais fraca.\end{solucao}
\begin{solucao}{11.31}\textbf{B.} Se filtros não usam a chave, pruning pode não ocorrer; a estratégia precisa refletir workload.\end{solucao}
\begin{solucao}{11.32}\textbf{A.} O problema é semântica temporal: event time, janelas e tratamento de atraso.\end{solucao}
\begin{solucao}{11.33}\textbf{CERTO.} ``Exactly once'' depende do limite da garantia; efeitos externos ainda podem requerer idempotência/transação.\end{solucao}
\begin{solucao}{11.34}\textbf{B.} Sem metadados, owner e qualidade, o lake tende a se tornar data swamp.\end{solucao}
\begin{solucao}{11.35}\textbf{A.} Entity resolution usa múltiplos sinais e trata ambiguidade; nome isolado é frágil.\end{solucao}
\begin{solucao}{11.36}\textbf{CERTO.} JOIN 1:N multiplica linhas e pode inflar medidas se a granularidade não for controlada.\end{solucao}
\begin{solucao}{11.37}\textbf{B.} Índices não usados podem custar escrita, mas remoção deve considerar dependências e sazonalidade.\end{solucao}
\begin{solucao}{11.38}\textbf{CERTO.} Baixa seletividade pode reduzir ganho de B-tree, mas outros índices/planos podem ser úteis.\end{solucao}
\begin{solucao}{11.39}\textbf{A.} Sem versionamento de código, parâmetros e dados, a alteração do indicador não é reproduzível.\end{solucao}
\begin{solucao}{11.40}\textbf{CERTO.} Quem altera o objeto e destrói a evidência fragiliza segregação e accountability.\end{solucao}

\begin{solucao}{11.41}\textbf{B.} Antes de ``corrigir'' o total, é preciso entender granularidade e cardinalidade, reconciliando cada estágio. DISTINCT pode apenas mascarar o problema.\end{solucao}
\begin{solucao}{11.42}\textbf{CERTO.} Desabilitar retry reduz resiliência; o desenho correto trata duplicidade.\end{solucao}
\begin{solucao}{11.43}\textbf{B.} Rejeição sem trilha destrói a capacidade de avaliar perda e legitimidade das regras.\end{solucao}
\begin{solucao}{11.44}\textbf{B.} WAL cria capacidade potencial, não evidência de restore. O teste valida cadeia, retenção e tempo.\end{solucao}
\begin{solucao}{11.45}\textbf{CERTO.} Disponibilidade técnica não garante qualidade analítica.\end{solucao}
\begin{solucao}{11.46}\textbf{B.} Ambiente interno não elimina need-to-know e mínimo privilégio.\end{solucao}
\begin{solucao}{11.47}\textbf{B.} O atributo só existe no futuro em relação ao momento da decisão: leakage temporal.\end{solucao}
\begin{solucao}{11.48}\textbf{CERTO.} Catálogo e lineage apoiam governança, mas não substituem validação e reconciliação.\end{solucao}
\begin{solucao}{11.49}\textbf{A.} Uma chave altamente desigual cria hotspot e compromete distribuição.\end{solucao}
\begin{solucao}{11.50}\textbf{C.} O achado correto foca reprodutibilidade e controles, não condena a ferramenta.\end{solucao}
