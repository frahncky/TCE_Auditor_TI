\section{Gabarito e resoluções comentadas — Engenharia de Software}

\begin{solucao}{12.01}\textbf{CERTO.} Iterativo significa refinar em ciclos; incremental significa acrescentar partes funcionais. Os dois atributos podem coexistir.\end{solucao}
\begin{solucao}{12.02}\textbf{ERRADO.} História é convite à conversa; critérios de aceitação tornam comportamento verificável e reduzem ambiguidade.\end{solucao}
\begin{solucao}{12.03}\textbf{CERTO.} A Daily é evento dos Developers para inspecionar progresso ao Sprint Goal e adaptar o Sprint Backlog.\end{solucao}
\begin{solucao}{12.04}\textbf{CERTO.} Limite de WIP reduz acúmulo e expõe gargalos, mas a variabilidade dos itens continua existindo.\end{solucao}
\begin{solucao}{12.05}\textbf{ERRADO.} TDD não substitui integração, sistema, segurança ou aceitação; cobertura alta também não prova suficiência.\end{solucao}
\begin{solucao}{12.06}\textbf{CERTO.} Cobertura mede execução estrutural, não qualidade dos asserts nem cobertura de requisitos.\end{solucao}
\begin{solucao}{12.07}\textbf{ERRADO.} Continuous delivery mantém software implantável; continuous deployment automatiza a promoção para produção.\end{solucao}
\begin{solucao}{12.08}\textbf{CERTO.} Readiness responde se pode receber tráfego; liveness responde se precisa ser reiniciada.\end{solucao}
\begin{solucao}{12.09}\textbf{ERRADO.} Média pode ocultar cauda severa. p95/p99 e jornadas críticas podem revelar experiência ruim.\end{solucao}
\begin{solucao}{12.10}\textbf{CERTO.} MVP reduz escopo de aprendizado, não obrigações essenciais.\end{solucao}

\begin{solucao}{12.11}\textbf{B.} Requisito não funcional precisa de métrica, limiar e contexto de carga.\end{solucao}
\begin{solucao}{12.12}\textbf{B.} Protótipo valida interação e entendimento, não arquitetura, segurança ou desempenho final.\end{solucao}
\begin{solucao}{12.13}\textbf{CERTO.} Sem DoD, o item não integra um Incremento utilizável conforme a definição acordada.\end{solucao}
\begin{solucao}{12.14}\textbf{B.} Pela Lei de Little, $W=L/\lambda=20/5=4$ dias.\end{solucao}
\begin{solucao}{12.15}\textbf{CERTO.} BDD é abordagem de colaboração e especificação de comportamento, não produto específico.\end{solucao}
\begin{solucao}{12.16}\textbf{B.} Valores imediatamente abaixo, nos limites e imediatamente acima exercitam as bordas do domínio.\end{solucao}
\begin{solucao}{12.17}\textbf{A.} Cobertura de linhas não garante cobertura de decisões/condições.\end{solucao}
\begin{solucao}{12.18}\textbf{CERTO.} Flakiness destrói confiança e aumenta custo de diagnóstico.\end{solucao}
\begin{solucao}{12.19}\textbf{B.} Integrar e testar só perto da release gera feedback tardio e alto custo de integração.\end{solucao}
\begin{solucao}{12.20}\textbf{B.} Credencial exposta deve ser considerada comprometida; remoção do último commit não a revoga.\end{solucao}
\begin{solucao}{12.21}\textbf{CERTO.} Promover o mesmo artefato reduz diferenças entre o que foi testado e o que foi implantado.\end{solucao}
\begin{solucao}{12.22}\textbf{B.} Tag mutável dificulta saber qual digest chegou a produção.\end{solucao}
\begin{solucao}{12.23}\textbf{B.} Startup probe protege aplicação lenta durante inicialização e evita loop de reinício prematuro.\end{solucao}
\begin{solucao}{12.24}\textbf{CERTO.} Service resolve descoberta/encaminhamento; persistência de estado é responsabilidade de outros componentes.\end{solucao}
\begin{solucao}{12.25}\textbf{C.} Error budget é $1-0,999=0,001=0,1\%$.\end{solucao}

\begin{solucao}{12.26}\textbf{C.} I e II estão corretas. III está errada: Daily não é aprovação pelo Product Owner.\end{solucao}
\begin{solucao}{12.27}\textbf{CERTO.} Se tudo é prioridade máxima, não há escolha explícita de trade-off.\end{solucao}
\begin{solucao}{12.28}\textbf{B.} Falhas estão nas interfaces; cobertura unitária adicional pode não capturá-las.\end{solucao}
\begin{solucao}{12.29}\textbf{B.} Cobertura e vulnerabilidade medem dimensões distintas; exceção precisa de decisão formal baseada em risco.\end{solucao}
\begin{solucao}{12.30}\textbf{CERTO.} Linhas de código podem incentivar complexidade desnecessária e gaming.\end{solucao}
\begin{solucao}{12.31}\textbf{B.} Dependência flutuante quebra determinismo do build.\end{solucao}
\begin{solucao}{12.32}\textbf{A.} Liveness não deve transformar indisponibilidade de dependência externa em reinício de processos saudáveis.\end{solucao}
\begin{solucao}{12.33}\textbf{CERTO.} Requests orientam scheduling; limits controlam consumo. Configuração ruim pode causar throttling/OOM.\end{solucao}
\begin{solucao}{12.34}\textbf{B.} p99 de 9 s mostra cauda severa mascarada pela média.\end{solucao}
\begin{solucao}{12.35}\textbf{A.} Trace/correlation ID liga uma mesma requisição através dos serviços.\end{solucao}
\begin{solucao}{12.36}\textbf{CERTO.} Canary reduz blast radius, mas exige observabilidade e rollback.\end{solucao}
\begin{solucao}{12.37}\textbf{A.} Feature flag separa deploy de exposição e permite rollout segmentado.\end{solucao}
\begin{solucao}{12.38}\textbf{B.} DevOps envolve cultura, fluxo, automação e feedback; Jenkins sozinho não basta.\end{solucao}
\begin{solucao}{12.39}\textbf{CERTO.} Rescheduling pode mover o workload; estado local não acompanha automaticamente o Pod.\end{solucao}
\begin{solucao}{12.40}\textbf{B.} Consumo rápido do error budget sinaliza reduzir mudanças e priorizar confiabilidade.\end{solucao}

\begin{solucao}{12.41}\textbf{ERRADO.} Automação sem segregação, gestão de segredos e proteção de logs apenas automatiza risco.\end{solucao}
\begin{solucao}{12.42}\textbf{A.} Sem cadeia requisito--mudança--teste--release, não se demonstra autorização nem validação.\end{solucao}
\begin{solucao}{12.43}\textbf{B.} Cobertura é um indicador; qualidade exige avaliar efetividade dos testes e defeitos reais.\end{solucao}
\begin{solucao}{12.44}\textbf{CERTO.} Mesmo commit pode gerar binário diferente se ambiente/dependências mudarem.\end{solucao}
\begin{solucao}{12.45}\textbf{A.} Privilégio e hostPath ampliam o impacto do workload comprometido para o host.\end{solucao}
\begin{solucao}{12.46}\textbf{CERTO.} Sem tempo consistente, a ordem causal fica ambígua.\end{solucao}
\begin{solucao}{12.47}\textbf{A.} A equipe otimiza frequência e ignora falhas/recuperação. Métricas precisam ser balanceadas.\end{solucao}
\begin{solucao}{12.48}\textbf{B.} MVP não suspende segurança, privacidade nem conformidade.\end{solucao}
\begin{solucao}{12.49}\textbf{CERTO.} SLI deve representar jornadas relevantes; agregado mensal pode esconder degradação localizada.\end{solucao}
\begin{solucao}{12.50}\textbf{C.} O achado correto descreve controles ausentes, risco e encaminhamento, sem impor ferramenta específica.\end{solucao}
