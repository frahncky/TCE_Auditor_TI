\section{Gabarito e resoluções comentadas — Gestão e Governança de TI}

\begin{solucao}{14.01}\textbf{Gabarito: A.} EDM reúne objetivos de governança. APO, BAI, DSS e MEA são domínios de gestão.\end{solucao}
\begin{solucao}{14.02}\textbf{Gabarito: A.} Governança avalia necessidades/opções, dirige prioridades e monitora resultados. As demais alternativas são atividades de gestão/operação.\end{solucao}
\begin{solucao}{14.03}\textbf{Gabarito: A.} APO = Align, Plan and Organize.\end{solucao}
\begin{solucao}{14.04}\textbf{Gabarito: A.} O Service Value System integra princípios, governança, cadeia de valor, práticas e melhoria contínua.\end{solucao}
\begin{solucao}{14.05}\textbf{Gabarito: A.} Incident management restaura serviço com rapidez e minimiza impacto. Investigar causa é foco de problem management.\end{solucao}
\begin{solucao}{14.06}\textbf{Gabarito: A.} Problem management atua sobre causas e erros conhecidos para reduzir recorrência/impacto.\end{solucao}
\begin{solucao}{14.07}\textbf{Gabarito: A.} PETI conecta TI à estratégia institucional e orienta objetivos de tecnologia.\end{solucao}
\begin{solucao}{14.08}\textbf{Gabarito: A.} KPI é indicador-chave de desempenho; os demais itens são de áreas técnicas diferentes.\end{solucao}
\begin{solucao}{14.09}\textbf{Gabarito: A.} Gateway exclusivo seleciona um caminho segundo condição. Paralelo ativa caminhos simultâneos.\end{solucao}
\begin{solucao}{14.10}\textbf{Gabarito: A.} DAMA-DMBOK organiza conhecimentos de gestão de dados: governança, qualidade, arquitetura, metadados etc.\end{solucao}
\begin{solucao}{14.11}\textbf{Gabarito: A.} Avaliar opções estratégicas, direcionar decisão e monitorar valor/risco é governança. Instalar, restaurar e corrigir são gestão/operação.\end{solucao}
\begin{solucao}{14.12}\textbf{Gabarito: A.} BAI = Build, Acquire and Implement.\end{solucao}
\begin{solucao}{14.13}\textbf{Gabarito: A.} MEA = Monitor, Evaluate and Assess.\end{solucao}
\begin{solucao}{14.14}\textbf{Gabarito: A.} Incidente recorrente indica necessidade de problem management para tratar causa, além da restauração feita por incident management.\end{solucao}
\begin{solucao}{14.15}\textbf{Gabarito: A.} Solicitação padrão e predefinida é service request. Não é necessariamente incidente ou problema.\end{solucao}
\begin{solucao}{14.16}\textbf{Gabarito: A.} Identificação, análise e avaliação formam risk assessment na ISO 31000. Tratamento vem depois.\end{solucao}
\begin{solucao}{14.17}\textbf{Gabarito: A.} Apetite expressa quanto/tipo de risco a organização aceita em busca de objetivos. Não é vulnerabilidade nem risco residual específico.\end{solucao}
\begin{solucao}{14.18}\textbf{Gabarito: A.} Sem owner, meta e indicador, o plano não se torna executável nem monitorável.\end{solucao}
\begin{solucao}{14.19}\textbf{Gabarito: A.} KPI governado precisa de semântica, fórmula, fonte, periodicidade, owner e interpretação/meta.\end{solucao}
\begin{solucao}{14.20}\textbf{Gabarito: A.} BSC traduz estratégia em objetivos/indicadores e relações entre perspectivas, adaptáveis ao setor público.\end{solucao}
\begin{solucao}{14.21}\textbf{Gabarito: A.} Sequence flow conecta elementos do mesmo processo/pool. Entre participantes usa-se message flow.\end{solucao}
\begin{solucao}{14.22}\textbf{Gabarito: A.} Message flow representa comunicação entre pools/participantes.\end{solucao}
\begin{solucao}{14.23}\textbf{Gabarito: A.} Compliance é aderência a obrigações aplicáveis. Não significa risco zero nem desempenho ótimo.\end{solucao}
\begin{solucao}{14.24}\textbf{Gabarito: A.} Governança de dados define decisão, autoridade, ownership, qualidade e acesso, não apenas tecnologia.\end{solucao}
\begin{solucao}{14.25}\textbf{Gabarito: A.} LAI e LGPD coexistem; divulgação exige avaliar transparência, finalidade, base legal, necessidade e restrições.\end{solucao}
\begin{solucao}{14.26}\textbf{Gabarito: A.} A mesma palavra pode aparecer nos dois níveis. Classificação depende de objeto e accountability, não do verbo isolado.\end{solucao}
\begin{solucao}{14.27}\textbf{Gabarito: A.} A cascata conecta necessidades dos stakeholders a objetivos e práticas, evitando controles sem relação com valor.\end{solucao}
\begin{solucao}{14.28}\textbf{Gabarito: A.} Tailoring adapta governança ao contexto e aos design factors. Copiar o framework integralmente ignora prioridade e risco.\end{solucao}
\begin{solucao}{14.29}\textbf{Gabarito: A.} Essas são as seis atividades da service value chain do ITIL 4.\end{solucao}
\begin{solucao}{14.30}\textbf{Gabarito: A.} Start where you are manda observar e medir o que já existe antes de descartar tudo.\end{solucao}
\begin{solucao}{14.31}\textbf{Gabarito: A.} Change enablement equilibra velocidade e risco para aumentar taxa de mudanças bem-sucedidas. Não é bloqueio universal.\end{solucao}
\begin{solucao}{14.32}\textbf{Gabarito: A.} Processo de projeto deve ser adaptado a porte, risco e incerteza. Burocracia uniforme gera desperdício.\end{solucao}
\begin{solucao}{14.33}\textbf{Gabarito: A.} O PMBOK 8 enfatiza valor, princípios, domínios e processos adaptáveis; não é simples repetição da estrutura da 6ª edição.\end{solucao}
\begin{solucao}{14.34}\textbf{Gabarito: A.} Métrica vira alvo e induz fechamento artificial. É exemplo de Goodhart/gaming.\end{solucao}
\begin{solucao}{14.35}\textbf{Gabarito: A.} Balanced metrics reduzem otimização local ao combinar velocidade, qualidade e resultado.\end{solucao}
\begin{solucao}{14.36}\textbf{Gabarito: A.} Gateway paralelo ativa/sincroniza múltiplos caminhos sem decisão exclusiva.\end{solucao}
\begin{solucao}{14.37}\textbf{Gabarito: A.} Pools representam participantes; fluxo de sequência não cruza pools. Comunicação usa message flow.\end{solucao}
\begin{solucao}{14.38}\textbf{Gabarito: A.} Conformidade é um piso em dado escopo. Implementação deficiente e risco residual permanecem possíveis.\end{solucao}
\begin{solucao}{14.39}\textbf{Gabarito: A.} Transparência não elimina proteção de dados. Deve haver análise jurídica e de minimização/divulgação.\end{solucao}
\begin{solucao}{14.40}\textbf{Gabarito: A.} Sem owner, ninguém responde claramente por definição, qualidade, acesso ou correção do dado.\end{solucao}
\begin{solucao}{14.41}\textbf{Gabarito: A.} Prazo é dimensão de entrega, não de valor. Governança deve acompanhar benefício, adoção, qualidade, risco e custo total.\end{solucao}
\begin{solucao}{14.42}\textbf{Gabarito: A.} Iniciativas sem vínculo estratégico podem consumir recurso sem demonstrar contribuição institucional.\end{solucao}
\begin{solucao}{14.43}\textbf{Gabarito: A.} Avaliar/direcionar sem monitorar benefícios fecha apenas parte do ciclo de governança.\end{solucao}
\begin{solucao}{14.44}\textbf{Gabarito: A.} KPI sem definição/fonte/owner não é reproduzível nem auditável. Quantidade de métricas não compensa baixa governança.\end{solucao}
\begin{solucao}{14.45}\textbf{Gabarito: A.} Métrica precisa de regra objetiva para janela, exclusões e fonte. Exclusão discricionária permite manipulação.\end{solucao}
\begin{solucao}{14.46}\textbf{Gabarito: A.} COBIT 2019 é desenhado para tailoring por fatores de desenho. Implementar tudo sem prioridade desperdiça recurso.\end{solucao}
\begin{solucao}{14.47}\textbf{Gabarito: A.} Apetite a risco é decisão de governança conectada aos objetivos institucionais, não apenas tema técnico de TI.\end{solucao}
\begin{solucao}{14.48}\textbf{Gabarito: A.} Melhoria contínua deve distinguir controles que reduzem risco de aprovações redundantes que só aumentam lead time.\end{solucao}
\begin{solucao}{14.49}\textbf{Gabarito: A.} Conflito semântico exige governança de definição: glossário, owner e decisão formal. Mais bancos aumentam o problema.\end{solucao}
\begin{solucao}{14.50}\textbf{Gabarito: A.} Certificação tem escopo e não garante ausência de incidentes. O auditor deve examinar operação e controles reais.\end{solucao}
