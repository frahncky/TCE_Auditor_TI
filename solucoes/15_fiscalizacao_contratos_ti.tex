\section{Gabarito e resoluções comentadas — Fiscalização de Contratos de TI}

\begin{solucao}{15.01}\textbf{Gabarito: ERRADO.} Três cotações não demonstram, por si, adequação da pesquisa. É necessário verificar independência, comparabilidade, contemporaneidade, parâmetros públicos e método. A pegadinha é transformar uma prática operacional em requisito suficiente.\end{solucao}
\begin{solucao}{15.02}\textbf{Gabarito: CERTO.} O ETP deve justificar a solução a partir da necessidade e das alternativas. Quando a solução já está escolhida, o estudo corre o risco de ser apenas racionalização posterior.\end{solucao}
\begin{solucao}{15.03}\textbf{Gabarito: ERRADO.} RTO contratual é objetivo, não evidência de capacidade. É preciso arquitetura, procedimentos e teste que demonstrem que o serviço pode voltar no prazo.\end{solucao}
\begin{solucao}{15.04}\textbf{Gabarito: CERTO.} Sem janela, fonte e exclusões, a mesma meta pode gerar resultados diferentes. A auditabilidade depende da definição operacional do indicador.\end{solucao}
\begin{solucao}{15.05}\textbf{Gabarito: ERRADO.} Ausência de penalidade pode significar bom desempenho, mas também fiscalização fraca, SLA mal desenhado ou descumprimentos não detectados. O auditor precisa de evidência positiva do cumprimento.\end{solucao}
\begin{solucao}{15.06}\textbf{Gabarito: CERTO.} Lock-in é risco de planejamento. Portabilidade e exit plan precisam ser negociados quando o órgão ainda possui poder de barganha.\end{solucao}
\begin{solucao}{15.07}\textbf{Gabarito: ERRADO.} Segregação é princípio de controle e deve ser proporcional ao risco. A interpretação absoluta de que ninguém pode participar de mais de uma fase é inadequada, especialmente em estruturas pequenas.\end{solucao}
\begin{solucao}{15.08}\textbf{Gabarito: ERRADO.} Se o objeto prevê resultados e níveis de serviço, presença de profissionais não prova entrega. O pagamento deve se relacionar ao resultado contratado.\end{solucao}
\begin{solucao}{15.09}\textbf{Gabarito: CERTO.} Uma pesquisa defensável precisa ser reproduzível: fontes, datas, ajustes, exclusões e fórmula devem estar documentados.\end{solucao}
\begin{solucao}{15.10}\textbf{Gabarito: ERRADO.} A matriz deve ser atualizada quando o contexto muda. Preservar a versão original é útil para histórico, mas congelar o risco durante a execução torna o controle ineficaz.\end{solucao}

\begin{solucao}{15.11}\textbf{Gabarito: B.} O dado disponível não sustenta 5.000 licenças. A análise deve considerar usuários ativos, simultaneidade, crescimento, perfis e possibilidade de contratação elástica. A alternativa D também é absoluta: simultaneidade pode não ser a única métrica de licenciamento.\end{solucao}
\begin{solucao}{15.12}\textbf{Gabarito: B.} Em 30 dias há 43.200 minutos; 0,1\% corresponde a 43,2 minutos. Com 70 minutos de indisponibilidade não excluída, a disponibilidade fica abaixo de 99,9\%.\end{solucao}
\begin{solucao}{15.13}\textbf{Gabarito: CERTO.} O ponto não é quem opera a ferramenta, mas se o órgão consegue validar completude, integridade e cálculo. Logs brutos protegidos e recálculo reduzem dependência de evidência unilateral.\end{solucao}
\begin{solucao}{15.14}\textbf{Gabarito: B.} Três propostas não justificam ignorar referências públicas significativamente menores. O auditor deve avaliar comparabilidade e possível viés da amostra.\end{solucao}
\begin{solucao}{15.15}\textbf{Gabarito: B.} ``Atividade realizada'' é afirmação, não evidência. Aceite deve demonstrar resultado por logs, testes, chamados, artefatos ou outro meio compatível com o objeto.\end{solucao}
\begin{solucao}{15.16}\textbf{Gabarito: ERRADO.} Mediana reduz sensibilidade a outliers, mas não corrige preços de objetos diferentes. Primeiro se assegura comparabilidade; depois se aplica método estatístico.\end{solucao}
\begin{solucao}{15.17}\textbf{Gabarito: B.} Comparar apenas licenças ignora custos relevantes de operação e transição. A escolha deve considerar TCO e risco, não preferência automática por SaaS ou on-premises.\end{solucao}
\begin{solucao}{15.18}\textbf{Gabarito: CERTO.} Ponto de função mede tamanho funcional, não qualidade, segurança, desempenho ou ausência de defeitos.\end{solucao}
\begin{solucao}{15.19}\textbf{Gabarito: B.} Não existe proibição automática. O auditor deve avaliar incompatibilidades, independência necessária e controles compensatórios. A questão explora a diferença entre princípio e regra absoluta.\end{solucao}
\begin{solucao}{15.20}\textbf{Gabarito: B.} Desconto de entrada pode ser anulado por custos de saída e dependência. Vantajosidade precisa considerar o ciclo de vida da solução.\end{solucao}
\begin{solucao}{15.21}\textbf{Gabarito: ERRADO.} Retenção de backup não demonstra restore, nem RTO. É necessário teste real de recuperação e evidência dos tempos obtidos.\end{solucao}
\begin{solucao}{15.22}\textbf{Gabarito: B.} Medir só resposta inicial cria gaming: o fornecedor responde rápido sem restaurar. Métricas devem representar o resultado que importa ao usuário.\end{solucao}
\begin{solucao}{15.23}\textbf{Gabarito: CERTO.} Prorrogação exige análise atual de desempenho e vantajosidade, não mera continuidade administrativa.\end{solucao}
\begin{solucao}{15.24}\textbf{Gabarito: B.} Uso para treinamento próprio é nova finalidade relevante e deve ser analisado sob LGPD, sigilo, propriedade e risco contratual. Criptografia não resolve finalidade inadequada.\end{solucao}
\begin{solucao}{15.25}\textbf{Gabarito: CERTO.} A estatística só é válida depois que se garante que os preços representam unidades e condições comparáveis.\end{solucao}

\begin{solucao}{15.26}\textbf{Gabarito: C.} I e II estão corretas. III está errada porque risco acompanha a execução e mudanças do contrato. SLA é apenas um mecanismo de controle, não substitui gerenciamento de riscos.\end{solucao}
\begin{solucao}{15.27}\textbf{Gabarito: CERTO.} Uma prova de conceito pode ser tecnicamente sofisticada e, ainda assim, estar enviesada por critérios desnecessários. Complexidade técnica não elimina risco de direcionamento.\end{solucao}
\begin{solucao}{15.28}\textbf{Gabarito: B.} Antes de pagar excedente, o fiscal precisa saber se o consumo era autorizado e necessário. Recursos sem tag/owner reduzem rastreabilidade e podem refletir desperdício.\end{solucao}
\begin{solucao}{15.29}\textbf{Gabarito: CERTO.} Evidência do fornecedor pode ser utilizada se houver controles de integridade e capacidade de validação. A banca pode tentar induzir ao extremo de considerar toda evidência interna ao fornecedor inválida.\end{solucao}
\begin{solucao}{15.30}\textbf{Gabarito: A.} Dimensionar pessoas sem definir cobertura e resultados pode remunerar presença, não capacidade de detecção e resposta.\end{solucao}
\begin{solucao}{15.31}\textbf{Gabarito: A.} Outlier incompatível pode ser excluído se a razão estiver documentada. A regra importante é análise crítica e transparência, não inclusão cega de todos os números.\end{solucao}
\begin{solucao}{15.32}\textbf{Gabarito: CERTO.} Menor preço trata preço de aquisição, não necessariamente custo total, utilização ou benefício. Economicidade é mais ampla.\end{solucao}
\begin{solucao}{15.33}\textbf{Gabarito: A.} Acesso privilegiado sem demanda ativa após encerramento é falha de offboarding e privilégio mínimo. Deve haver revogação ou autorização excepcional documentada.\end{solucao}
\begin{solucao}{15.34}\textbf{Gabarito: CERTO.} SLA precisa ser tecnicamente atingível. Um SPOF conhecido pode tornar a meta inconsistente ou exigir análise de risco/custo não demonstrada.\end{solucao}
\begin{solucao}{15.35}\textbf{Gabarito: C.} Os três sinais podem indicar barreira indevida, mas não provam fraude isoladamente. Eles justificam testes adicionais de necessidade, proporcionalidade e competição.\end{solucao}
\begin{solucao}{15.36}\textbf{Gabarito: ERRADO.} Recompilar pode gerar binário diferente por dependências, ferramentas ou ambiente. Promover o mesmo artefato testado oferece rastreabilidade mais forte.\end{solucao}
\begin{solucao}{15.37}\textbf{Gabarito: B.} Critério de segurança explicitamente previsto integra o aceite. Cumprimento funcional não autoriza ignorar não conformidade crítica.\end{solucao}
\begin{solucao}{15.38}\textbf{Gabarito: CERTO.} Encerramentos brutos podem ser manipulados. Catálogo, severidade, aceite e reabertura aproximam a medição do resultado real.\end{solucao}
\begin{solucao}{15.39}\textbf{Gabarito: C.} Reequilíbrio exige análise jurídica e factual. Não é automático, nem proibido por categoria de contrato.\end{solucao}
\begin{solucao}{15.40}\textbf{Gabarito: CERTO.} Mesmo preço unitário pode tornar-se desvantajoso se quantidade contratada deixa de refletir demanda. A economicidade é avaliada no contexto atual.\end{solucao}

\begin{solucao}{15.41}\textbf{Gabarito: ERRADO.} Os dados fornecidos mostram exatamente que a ausência de penalidades não é prova de desempenho. Falta trilha entre serviço, aceite e pagamento. O procedimento correto é reconstruir a cadeia de evidências.\end{solucao}
\begin{solucao}{15.42}\textbf{Gabarito: B.} Com oito meses restantes, a prioridade é reduzir risco de saída: testar exportação, dependências, chaves e cronograma. Negociar preço sem capacidade de migrar perpetua lock-in.\end{solucao}
\begin{solucao}{15.43}\textbf{Gabarito: B.} O cenário sugere superdimensionamento, mas o auditor não deve saltar diretamente para dano. É preciso comparar projeções originais, alternativas e custo evitável.\end{solucao}
\begin{solucao}{15.44}\textbf{Gabarito: CERTO.} Suficiência e confiabilidade dependem de controle, integridade e possibilidade de corroboração. Origem na contratada aumenta risco de viés, mas não invalida automaticamente a evidência.\end{solucao}
\begin{solucao}{15.45}\textbf{Gabarito: B.} A cláusula cria risco de uso secundário, retenção excessiva, memorização e exposição de informação. É um problema de governança de dados e contrato, não apenas de infraestrutura.\end{solucao}
\begin{solucao}{15.46}\textbf{Gabarito: ERRADO.} O fiscal não pode flexibilizar informalmente critério material para atender conveniência orçamentária. Alterações de aceite precisam seguir o regime contratual e não podem mascarar não conformidade.\end{solucao}
\begin{solucao}{15.47}\textbf{Gabarito: B.} O problema está na integridade e independência da medição. Replicar telemetria e preservar logs sob controle do órgão permite recalcular SLA e detectar alteração.\end{solucao}
\begin{solucao}{15.48}\textbf{Gabarito: C.} Backup diário, sem mecanismo de captura de alterações a cada 15 min, não demonstra RPO de 15 min. Restore de 5 h também não demonstra RTO de 2 h.\end{solucao}
\begin{solucao}{15.49}\textbf{Gabarito: CERTO.} Regularidade formal não exclui ineficiência ou desperdício. Quantitativos, utilização e benefício integram a análise de economicidade.\end{solucao}
\begin{solucao}{15.50}\textbf{Gabarito: C.} Um bom achado descreve condição, causa/controle ausente, efeito e encaminhamento. As demais alternativas são conclusões amplas, prescritivas ou sem base suficiente.\end{solucao}
