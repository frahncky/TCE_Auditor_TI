\section{Banco de questões — Engenharia de Software}

\subsection*{N1 — Fundamentos}
\begin{questao}{\nivel{N1} 12.01}Um processo incremental caracteriza-se por:\begin{enumerate}[label=\Alph*)]\item entregar parcelas funcionais sucessivas do produto\item proibir revisões\item exigir entrega única ao final\item eliminar testes\item impedir mudança de requisitos\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 12.02}Um requisito funcional descreve principalmente:\begin{enumerate}[label=\Alph*)]\item comportamento ou serviço que o sistema deve oferecer\item somente tempo de resposta\item apenas tecnologia de banco\item orçamento do projeto\item equipe necessária\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 12.03}Qual artefato do Scrum contém trabalho selecionado para a Sprint e plano para entregá-lo?\begin{enumerate}[label=\Alph*)]\item Sprint Backlog\item Product Goal\item Increment apenas\item Kanban board obrigatoriamente\item Definition of Ready obrigatória pelo Scrum Guide\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 12.04}Em TDD, o ciclo clássico é:\begin{enumerate}[label=\Alph*)]\item teste falha, código mínimo passa, refatoração\item deploy, backup, rollback\item requisito, contrato, licitação\item análise, desenho, operação sem teste\item build, monitoramento, incidente\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 12.05}Teste unitário visa principalmente:\begin{enumerate}[label=\Alph*)]\item verificar unidade pequena de código em isolamento adequado\item validar toda infraestrutura de produção\item testar apenas usabilidade\item substituir teste de integração\item medir SLA contratual\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 12.06}Análise estática de código é realizada:\begin{enumerate}[label=\Alph*)]\item sem executar o programa, examinando código/artefatos\item apenas em produção\item somente por usuários finais\item necessariamente com carga máxima\item após apagar o código-fonte\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 12.07}Em CI/CD, integração contínua enfatiza:\begin{enumerate}[label=\Alph*)]\item integrar mudanças frequentemente com build/testes automatizados\item implantar manualmente uma vez ao ano\item remover controle de versão\item impedir feedback\item usar apenas containers\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 12.08}No Kubernetes, a menor unidade implantável que agrupa um ou mais contêineres é:\begin{enumerate}[label=\Alph*)]\item Pod\item NodePool\item ClusterRole\item Namespace\item Ingress\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 12.09}Em observabilidade, os três sinais clássicos são:\begin{enumerate}[label=\Alph*)]\item métricas, logs e traces\item CPU, teclado e mouse\item DNS, DHCP e NTP\item commit, merge e tag\item usuário, senha e token\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 12.10}MVP significa:\begin{enumerate}[label=\Alph*)]\item menor produto viável para testar hipótese relevante de valor/aprendizado\item produto final sem requisitos\item protótipo que jamais chega a usuário\item versão sem segurança obrigatoriamente\item máximo volume de processo\end{enumerate}\end{questao}

\subsection*{N2 — Aplicação}
\begin{questao}{\nivel{N2} 12.11}A história ``Como auditor, quero filtrar contratos por risco para priorizar fiscalização'' deve ser complementada principalmente por:\begin{enumerate}[label=\Alph*)]\item critérios de aceitação verificáveis\item apenas nome do desenvolvedor\item endereço IP do servidor\item RAID usado\item contrato de trabalho\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.12}Um requisito ``95\% das consultas devem responder em até 2 s'' é:\begin{enumerate}[label=\Alph*)]\item não funcional e mensurável\item funcional exclusivamente\item história sem critério\item regra de banco apenas\item requisito impossível de testar\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.13}Em Scrum, a Daily Scrum tem por objetivo principal:\begin{enumerate}[label=\Alph*)]\item inspecionar progresso ao Sprint Goal e adaptar plano de trabalho\item prestar contas ao Scrum Master\item aprovar orçamento anual\item substituir Sprint Review\item revisar contratos jurídicos\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.14}Kanban limita WIP principalmente para:\begin{enumerate}[label=\Alph*)]\item reduzir acúmulo, expor gargalos e melhorar fluxo\item aumentar multitarefa indefinidamente\item eliminar priorização\item impedir medição de lead time\item substituir requisitos\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.15}BDD busca principalmente:\begin{enumerate}[label=\Alph*)]\item descrever comportamento observável em linguagem compartilhada por negócio e tecnologia\item medir CPU\item substituir todo teste unitário\item criar banco dimensional\item administrar firewall\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.16}ATDD diferencia-se por enfatizar:\begin{enumerate}[label=\Alph*)]\item critérios/testes de aceitação definidos colaborativamente antes da implementação\item testes somente após produção\item ausência de usuário\item código sem requisito\item deploy manual\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.17}Uma equipe tem 100\% de cobertura de linhas, mas defeitos continuam em produção. A conclusão correta é:\begin{enumerate}[label=\Alph*)]\item cobertura mede execução, não suficiência dos cenários nem qualidade dos asserts\item cobertura prova ausência de defeitos\item testes devem ser removidos\item só teste de carga detecta regra errada\item SonarQube elimina necessidade de testes\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.18}Teste de integração verifica principalmente:\begin{enumerate}[label=\Alph*)]\item interação entre componentes/interfaces\item apenas método isolado\item somente experiência visual\item política de férias\item arquitetura física do datacenter\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.19}Em teste caixa-preta, uma técnica típica é:\begin{enumerate}[label=\Alph*)]\item classes de equivalência e valores-limite\item cobertura de decisão interna apenas\item análise de complexidade ciclomática exclusivamente\item inspeção de assembly\item profiling de CPU somente\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.20}Em teste caixa-branca, uma métrica típica é:\begin{enumerate}[label=\Alph*)]\item cobertura de decisões/branches\item satisfação do usuário\item NPS\item SLA de fornecedor\item disponibilidade elétrica\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.21}Entrega contínua significa que o software:\begin{enumerate}[label=\Alph*)]\item permanece em estado implantável com automação, mas produção pode exigir decisão/aprovação\item é obrigatoriamente implantado em produção a cada commit\item não possui testes\item dispensa versionamento\item só pode ser monolítico\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.22}No Docker, uma imagem é melhor descrita como:\begin{enumerate}[label=\Alph*)]\item artefato imutável/em camadas usado para criar contêineres\item processo em execução apenas\item volume persistente\item cluster Kubernetes\item firewall\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.23}Em Kubernetes, um Service oferece principalmente:\begin{enumerate}[label=\Alph*)]\item endpoint estável e descoberta/balanceamento para Pods selecionados\item armazenamento de segredo necessariamente\item compilação de código\item controle de versão\item banco relacional\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.24}Readiness probe indica:\begin{enumerate}[label=\Alph*)]\item se o workload está apto a receber tráfego\item se deve ser reiniciado por estar morto necessariamente\item se o código compilou\item se o banco está normalizado\item se o usuário autenticou\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 12.25}SLI, SLO e SLA diferem porque:\begin{enumerate}[label=\Alph*)]\item SLI mede; SLO define objetivo; SLA formaliza compromisso/acordo com consequências conforme caso\item são sinônimos\item SLI é contrato e SLA é métrica\item SLO mede hardware\item SLA nunca contém alvo\end{enumerate}\end{questao}

\subsection*{N3 — Nível de concurso}
\begin{questao}{\nivel{N3} 12.26}Um protótipo de interface foi validado por usuários. A conclusão correta é:\begin{enumerate}[label=\Alph*)]\item reduz incerteza de interação, mas não prova arquitetura, desempenho ou segurança da solução final\item já é o produto final\item substitui requisitos não funcionais\item elimina testes\item garante escalabilidade\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.27}Um Product Owner transforma toda demanda em prioridade máxima. Qual consequência provável?\begin{enumerate}[label=\Alph*)]\item ausência de priorização real e dispersão de capacidade\item aumento automático de valor\item redução garantida de lead time\item eliminação de risco\item cumprimento obrigatório do Sprint Goal\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.28}A Lei de Little em sistema estável relaciona WIP, throughput e lead time. Se WIP médio aumenta sem aumento de throughput, tende-se a observar:\begin{enumerate}[label=\Alph*)]\item aumento do tempo médio no sistema\item lead time zero\item redução obrigatória de filas\item throughput infinito\item nenhuma relação\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.29}Uma suíte E2E leva 4 horas e falha de forma intermitente. Qual melhoria é mais adequada?\begin{enumerate}[label=\Alph*)]\item aumentar testes rápidos em níveis inferiores, estabilizar E2E críticos e atacar flakiness\item eliminar todos os unitários\item aceitar flakiness\item executar somente mensalmente sem feedback\item substituir testes por cobertura\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.30}Complexidade ciclomática elevada indica principalmente:\begin{enumerate}[label=\Alph*)]\item maior número de caminhos independentes e potencial dificuldade de teste/manutenção\item maior disponibilidade\item menor necessidade de teste\item ausência de branches\item qualidade garantida\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.31}Um quality gate do SonarQube falha por vulnerabilidade crítica. Qual prática de CI é mais coerente?\begin{enumerate}[label=\Alph*)]\item interromper promoção até avaliação/correção ou exceção formal baseada em risco\item ignorar sempre o gate\item apagar resultado\item promover e desabilitar logs\item remover análise estática\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.32}Pipeline compila dependência externa por versão flutuante, e builds idênticos geram artefatos diferentes em dias distintos. O problema é:\begin{enumerate}[label=\Alph*)]\item baixa reprodutibilidade da cadeia de build\item excesso de cobertura\item problema de VLAN\item falta de normalização\item RPO inadequado\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.33}Uma credencial administrativa está hard-coded no repositório Git. A ação prioritária é:\begin{enumerate}[label=\Alph*)]\item revogar/rotacionar a credencial e migrar para gestão de segredos, além de tratar histórico\item apenas ocultar a linha no branch atual\item aumentar tamanho da senha sem rotacionar\item mover o arquivo para outra pasta\item confiar no .gitignore retroativamente\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.34}Um Deployment Kubernetes usa tag \texttt{latest}. Qual risco principal?\begin{enumerate}[label=\Alph*)]\item baixa determinismo/reprodutibilidade sobre qual imagem foi implantada\item impossibilidade de usar rede\item ausência de Pods\item proibição de rollback\item falta de DNS obrigatória\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.35}Uma liveness probe depende de serviço externo temporariamente indisponível. Isso pode causar:\begin{enumerate}[label=\Alph*)]\item reinícios desnecessários e amplificação da indisponibilidade\item melhoria garantida de resiliência\item redução de tráfego externo\item normalização de banco\item aumento de cobertura\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.36}Uma aplicação mantém sessão somente na memória local de cada Pod e usa autoscaling. Para evitar perda de sessão, deve-se:\begin{enumerate}[label=\Alph*)]\item externalizar estado ou adotar mecanismo stateless/replicado compatível\item fixar um único Pod para sempre\item remover balanceamento\item desativar health checks\item usar RAID 0\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.37}Em observabilidade distribuída, correlation/trace ID é útil para:\begin{enumerate}[label=\Alph*)]\item seguir uma requisição por múltiplos serviços\item substituir autenticação\item criptografar banco\item criar índice SQL\item definir Sprint Goal\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.38}Um sistema cumpre latência média de 100 ms, mas p99 é 8 s. Qual conclusão?\begin{enumerate}[label=\Alph*)]\item média esconde cauda severa; percentis devem ser considerados no SLO\item desempenho é ótimo por definição\item p99 é irrelevante\item média sempre representa todos usuários\item deve-se medir apenas CPU\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.39}Canary deployment reduz risco porque:\begin{enumerate}[label=\Alph*)]\item expõe nova versão gradualmente a parcela do tráfego e permite observar métricas antes da expansão\item elimina necessidade de rollback\item impede qualquer defeito\item substitui testes\item exige downtime total\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 12.40}Feature flag permite:\begin{enumerate}[label=\Alph*)]\item separar deploy de ativação de funcionalidade e controlar exposição\item eliminar versionamento\item remover código antigo automaticamente\item substituir autorização\item dispensar testes\end{enumerate}\end{questao}

\subsection*{N4 — Avançado / Auditoria}
\begin{questao}{\nivel{N4} 12.41}Auditoria identifica deploy automático em produção, segredo fixo no repositório, ausência de revisão para mudanças críticas e logs apagáveis pelo mesmo usuário que faz deploy. A recomendação mais completa é:\begin{enumerate}[label=\Alph*)]\item gestão de segredos, segregação/aprovação baseada em risco, trilha protegida e identidade nominativa\item aumentar apenas cobertura\item remover automação totalmente\item usar FTP\item manter segredo e aumentar complexidade\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 12.42}Uma equipe mede produtividade por linhas de código produzidas. Qual risco gerencial?\begin{enumerate}[label=\Alph*)]\item incentivo perverso a aumentar código em vez de valor, qualidade e simplicidade\item métrica perfeita de valor\item elimina refatoração\item garante segurança\item mede satisfação do usuário diretamente\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 12.43}Um fornecedor apresenta 85\% de cobertura como prova de qualidade. O auditor deve:\begin{enumerate}[label=\Alph*)]\item examinar critérios, asserts, níveis de teste, defeitos, mutação/efetividade e requisitos críticos, pois cobertura isolada é insuficiente\item aceitar sem testes adicionais\item exigir 100\% e encerrar auditoria\item ignorar qualquer cobertura\item substituir por contagem de commits\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 12.44}Um sistema crítico não possui rastreabilidade entre requisito, commit, teste e release. Qual risco?\begin{enumerate}[label=\Alph*)]\item dificuldade de demonstrar que mudanças autorizadas implementaram requisitos e foram testadas\item excesso de documentação\item aumento automático de segurança\item impossibilidade de usar Git\item somente problema visual\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 12.45}O cluster Kubernetes permite workloads privilegiados e hostPath para qualquer equipe. Qual risco?\begin{enumerate}[label=\Alph*)]\item escape/comprometimento do nó e ampliação de impacto por privilégio excessivo\item somente latência de rede\item falta de requisito funcional\item nenhuma, pois contêiner é VM\item apenas problema de UI\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 12.46}Uma pipeline usa artefato recompilado separadamente para homologação e produção. Qual fragilidade?\begin{enumerate}[label=\Alph*)]\item produção pode receber binário diferente do testado; ideal é promover o mesmo artefato imutável\item melhora rastreabilidade\item elimina supply-chain risk\item garante reproducibilidade\item não há diferença\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 12.47}Um incidente grave ocorreu, mas logs de aplicação não possuem timestamp sincronizado e cada serviço usa fuso diferente. Qual consequência?\begin{enumerate}[label=\Alph*)]\item dificuldade de reconstruir sequência causal e correlacionar evidências\item melhora privacidade\item elimina necessidade de traces\item impede qualquer autenticação\item aumenta throughput\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 12.48}Um SLO é 99,9\% de disponibilidade mensal. A equipe mantém 99,95\%, mas consome todo error budget nos primeiros dias. Uma interpretação madura é:\begin{enumerate}[label=\Alph*)]\item observar tendência e error budget para equilibrar mudança e confiabilidade, não apenas média final\item continuar deploys arriscados sem limite\item SLO não serve para operação\item error budget é orçamento financeiro\item disponibilidade não deve ser medida\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 12.49}Uma organização chama de ``DevOps'' apenas a adoção de Jenkins, sem integração entre equipes, automação confiável ou feedback operacional. A avaliação correta é:\begin{enumerate}[label=\Alph*)]\item ferramenta isolada não caracteriza transformação DevOps; cultura, fluxo, automação e feedback são centrais\item Jenkins por si garante DevOps\item DevOps significa eliminar operações\item DevOps proíbe controles\item DevOps é sinônimo de Kubernetes\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 12.50}Um fornecedor entrega MVP de sistema público sem controles mínimos de proteção de dados, alegando que ``segurança fica para depois''. A conclusão é:\begin{enumerate}[label=\Alph*)]\item MVP não autoriza descumprir requisitos essenciais de segurança, privacidade e conformidade\item correta, pois MVP dispensa qualquer qualidade\item correta se houver poucos usuários\item segurança só existe após versão 1.0\item privacidade não é requisito de software\end{enumerate}\end{questao}
