\section{Banco de questões — Engenharia de Dados}

\subsection*{N1 — Fundamentos}

\begin{questao}{\nivel{N1} 11.01}No modelo relacional, uma chave primária tem como função principal:\begin{enumerate}[label=\Alph*)]\item identificar unicamente cada tupla\item ordenar fisicamente toda tabela por definição\item substituir qualquer índice\item permitir valores duplicados obrigatoriamente\item eliminar a necessidade de integridade referencial\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 11.02}Uma chave estrangeira é usada principalmente para:\begin{enumerate}[label=\Alph*)]\item representar referência entre relações e apoiar integridade referencial\item criptografar colunas\item criar backup\item definir tamanho de página\item substituir normalização\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 11.03}A Primeira Forma Normal busca, em abordagem clássica, garantir:\begin{enumerate}[label=\Alph*)]\item atributos com valores atômicos no contexto do modelo e ausência de grupos repetitivos\item ausência de qualquer redundância\item toda relação em BCNF\item somente chaves simples\item uso obrigatório de surrogate key\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 11.04}Em SQL, qual comando pertence tipicamente à DDL?\begin{enumerate}[label=\Alph*)]\item CREATE TABLE\item SELECT\item UPDATE\item DELETE\item INSERT\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 11.05}Em um esquema dimensional, a tabela que normalmente registra medidas na granularidade definida é:\begin{enumerate}[label=\Alph*)]\item fato\item dimensão\item catálogo\item índice\item view materializada necessariamente\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 11.06}Qual alternativa é exemplo típico de banco NoSQL orientado a documentos?\begin{enumerate}[label=\Alph*)]\item armazenamento de objetos JSON/BSON por documento\item tabela relacional em 3FN exclusivamente\item planilha CSV sem mecanismo de consulta\item RAID 5\item DNS autoritativo\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 11.07}ETL significa:\begin{enumerate}[label=\Alph*)]\item Extract, Transform, Load\item Execute, Test, Link\item Encode, Transfer, Log\item Extract, Transfer, Lock\item Evaluate, Transform, Learn\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 11.08}No contexto de dados, linhagem (data lineage) registra principalmente:\begin{enumerate}[label=\Alph*)]\item origem, transformações e destinos dos dados\item somente a senha do DBA\item endereços IP da aplicação\item quantidade de CPU do banco\item permissões de arquivos do sistema operacional\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 11.09}Qual propriedade ACID trata da permanência dos efeitos de uma transação confirmada?\begin{enumerate}[label=\Alph*)]\item Durabilidade\item Atomicidade\item Consistência\item Isolamento\item Idempotência\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 11.10}Em SQL, a cláusula HAVING é usada tipicamente para:\begin{enumerate}[label=\Alph*)]\item filtrar grupos após agregação\item criar tabela\item definir chave estrangeira\item ordenar fisicamente páginas\item iniciar transação distribuída\end{enumerate}\end{questao}

\subsection*{N2 — Aplicação}

\begin{questao}{\nivel{N2} 11.11}Uma relação tem chave composta $(A,B)$ e atributo $C$ depende apenas de $A$. O problema é uma dependência:\begin{enumerate}[label=\Alph*)]\item parcial, tratada pela 2FN\item multivalorada apenas\item transitiva obrigatoriamente\item de junção\item inexistente\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.12}Uma tabela \texttt{Pedido(PedidoID, ClienteID, NomeCliente)} possui \texttt{PedidoID} como chave e \texttt{NomeCliente} depende de \texttt{ClienteID}, que não é chave. Isso sugere:\begin{enumerate}[label=\Alph*)]\item dependência transitiva e potencial violação de 3FN\item violação de 1FN apenas\item ausência de dependência funcional\item necessidade de RAID\item problema de rede\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.13}Para listar todos os órgãos, inclusive os que não possuem contratos, preservando a tabela \texttt{orgao}, a junção mais natural é:\begin{enumerate}[label=\Alph*)]\item LEFT JOIN\item INNER JOIN\item CROSS JOIN sem condição\item NATURAL JOIN obrigatoriamente\item DELETE JOIN\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.14}Qual índice tende a ser útil para consultas por igualdade e intervalo em coluna ordenável, em SGBDs relacionais comuns?\begin{enumerate}[label=\Alph*)]\item B-tree\item somente hash\item RAID 0\item bitmap em qualquer workload obrigatoriamente\item nenhum índice pode atender intervalos\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.15}Uma coluna possui milhões de linhas e apenas dois valores possíveis. Qual afirmação é mais correta?\begin{enumerate}[label=\Alph*)]\item baixa cardinalidade pode reduzir seletividade de certos índices tradicionais; o plano depende do SGBD e da consulta\item índice B-tree sempre torna qualquer consulta mais rápida\item cardinalidade não influencia otimização\item índice elimina custo de escrita\item o banco deixa de ser relacional\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.16}Em modelo dimensional, a granularidade deve ser definida antes das medidas porque:\begin{enumerate}[label=\Alph*)]\item determina o que cada linha da tabela fato representa\item define o endereço IP do DW\item substitui dimensões\item determina obrigatoriamente o SGBD\item elimina necessidade de ETL\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.17}Uma medida de saldo de conta pode ser somada entre contas, mas não necessariamente ao longo de datas. Ela é tipicamente:\begin{enumerate}[label=\Alph*)]\item semi-aditiva\item totalmente aditiva em todas as dimensões\item não numérica\item chave degenerada\item dimensão lentamente mutável\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.18}Em ELT, a diferença central em relação a ETL é:\begin{enumerate}[label=\Alph*)]\item carregar dados antes de executar parte relevante das transformações no ambiente de destino\item eliminar extração\item não usar armazenamento\item proibir transformação\item exigir banco relacional\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.19}Um pipeline recebe o mesmo evento duas vezes após retry e deve evitar efeito duplicado. A propriedade desejada é:\begin{enumerate}[label=\Alph*)]\item idempotência\item cardinalidade\item normalização BCNF\item compressão\item serialização física\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.20}Qual dimensão de qualidade verifica se campos obrigatórios estão presentes?\begin{enumerate}[label=\Alph*)]\item completude\item unicidade exclusivamente\item atualidade\item consistência semântica apenas\item latência\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.21}Em uma transação bancária, debitar uma conta e creditar outra devem ocorrer juntos ou nenhum ocorrer. Isso exemplifica:\begin{enumerate}[label=\Alph*)]\item atomicidade\item isolamento apenas\item durabilidade apenas\item disponibilidade\item particionamento\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.22}Leitura de dado ainda não confirmado por outra transação é denominada:\begin{enumerate}[label=\Alph*)]\item dirty read\item phantom write\item deadlock obrigatório\item full scan\item cache miss\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.23}Uma consulta possui filtro em coluna sem índice, retornando 0,01\% de tabela de 500 milhões de linhas. Qual ação deve ser avaliada primeiro?\begin{enumerate}[label=\Alph*)]\item examinar plano, estatísticas e possibilidade de índice adequado ao padrão de acesso\item adicionar índice em todas as colunas sem análise\item normalizar para 1FN novamente necessariamente\item trocar SQL por CSV\item eliminar filtro\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.24}Em um data lake, catálogo de metadados é importante porque:\begin{enumerate}[label=\Alph*)]\item facilita descoberta, significado, esquema, ownership e governança dos conjuntos\item substitui backup\item cifra todos os dados automaticamente\item impede qualquer dado duplicado\item elimina necessidade de qualidade\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 11.25}Em integração via arquivos, um controle básico para detectar carga incompleta é:\begin{enumerate}[label=\Alph*)]\item reconciliar contagem, tamanho/hash e totais de controle entre origem e destino\item confiar apenas no nome do arquivo\item remover logs\item aceitar qualquer encoding\item impedir reprocessamento\end{enumerate}\end{questao}

\subsection*{N3 — Nível de concurso}

\begin{questao}{\nivel{N3} 11.26}Uma tabela tem chave composta $(AlunoID,DisciplinaID)$ e \texttt{NomeAluno} depende de \texttt{AlunoID}. A decomposição correta deve:\begin{enumerate}[label=\Alph*)]\item separar dados do aluno para remover dependência parcial\item duplicar NomeAluno em cada matrícula para melhorar integridade\item eliminar DisciplinaID\item converter tudo para JSON obrigatoriamente\item adicionar índice sem alterar modelo\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.27}Uma consulta usa \texttt{LEFT JOIN}, mas coloca em \texttt{WHERE} condição \texttt{t2.status='ATIVO'} sobre a tabela direita, sem aceitar NULL. O efeito provável é:\begin{enumerate}[label=\Alph*)]\item eliminar linhas sem correspondência, aproximando o resultado de INNER JOIN para esse predicado\item preservar sempre todas as linhas da esquerda\item gerar CROSS JOIN\item impedir qualquer uso de índice\item transformar a consulta em DDL\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.28}Uma aplicação sofre lost update entre transações concorrentes. Qual tema deve ser avaliado?\begin{enumerate}[label=\Alph*)]\item isolamento/controle de concorrência e padrão de atualização\item normalização de dimensões apenas\item RAID\item DNS\item compressão de backup\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.29}MVCC é usado por diversos SGBDs para:\begin{enumerate}[label=\Alph*)]\item manter múltiplas versões e reduzir bloqueios entre leitores e escritores conforme isolamento\item substituir WAL e backup em todos os casos\item tornar transações não atômicas\item eliminar necessidade de vacuum/limpeza em qualquer implementação\item impedir índices\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.30}Um índice composto é \texttt{(orgao_id, data)}. Em regra, qual consulta tende a aproveitar melhor o prefixo do índice?\begin{enumerate}[label=\Alph*)]\item filtro por \texttt{orgao_id} e intervalo de \texttt{data}\item filtro somente por coluna não relacionada\item consulta sem predicado sempre\item filtro apenas por expressão não sargable obrigatoriamente\item nenhuma consulta pode usar índice composto\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.31}Uma tabela fato registra uma linha por item de nota fiscal. Somar \texttt{valor_total_nota} repetido em cada item causa:\begin{enumerate}[label=\Alph*)]\item dupla contagem por incompatibilidade entre medida e granularidade\item normalização excessiva apenas\item melhoria de precisão\item eliminação de dimensão\item consistência automática\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.32}Em slowly changing dimension tipo 2, uma alteração de atributo histórico é tratada tipicamente por:\begin{enumerate}[label=\Alph*)]\item nova linha/versionamento com período de validade\item sobrescrever sempre sem histórico\item apagar a dimensão\item atualizar a tabela fato apenas\item converter chave para hash sem versão\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.33}No teorema CAP, durante uma partição de rede, um sistema distribuído não consegue garantir simultaneamente:\begin{enumerate}[label=\Alph*)]\item consistência forte e disponibilidade para todas as requisições\item persistência e criptografia\item SQL e NoSQL\item backup e replicação\item compressão e deduplicação\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.34}Um banco chave-valor é especialmente natural quando:\begin{enumerate}[label=\Alph*)]\item o principal padrão de acesso é recuperar valor por chave conhecida com esquema simples\item são necessárias junções relacionais complexas como requisito central\item toda consulta é OLAP dimensional\item não existe chave\item é obrigatório usar SQL ANSI completo\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.35}Em streaming, ``exactly once'' deve ser interpretado com cuidado porque:\begin{enumerate}[label=\Alph*)]\item a garantia depende do limite do sistema e dos efeitos externos; idempotência continua relevante\item significa que a rede nunca retransmite\item elimina falhas\item dispensa transações\item impede duplicatas em qualquer sistema externo automaticamente\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.36}Um pipeline usa timestamp de processamento para agrupar eventos que chegam atrasados. Para refletir quando o evento realmente ocorreu, deve-se considerar:\begin{enumerate}[label=\Alph*)]\item event time, janelas e política para dados atrasados/watermarks\item somente CPU do consumidor\item RAID 6\item DNS TTL\item chave primária numérica\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.37}Um data warehouse recebe dados de fornecedores com CNPJ em formatos diferentes. Qual controle de qualidade é mais apropriado?\begin{enumerate}[label=\Alph*)]\item padronizar, validar domínio/formato, deduplicar e manter regra de transformação rastreável\item converter CNPJ em endereço IP\item apagar registros divergentes sem evidência\item substituir por nome fantasia\item aceitar qualquer valor\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.38}Em otimização SQL, uma função aplicada diretamente à coluna indexada no predicado pode tornar o filtro não sargable. A consequência possível é:\begin{enumerate}[label=\Alph*)]\item dificultar uso eficiente do índice, dependendo do SGBD e de índices funcionais disponíveis\item garantir index seek\item eliminar custo de CPU\item converter consulta em transação serializável\item impedir qualquer resultado\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.39}Um índice acelera SELECTs, mas aumenta custo de INSERT/UPDATE/DELETE porque:\begin{enumerate}[label=\Alph*)]\item estruturas de índice também precisam ser mantidas\item o índice elimina páginas de dados\item todo índice duplica o banco integralmente\item índices impedem cache\item índices removem estatísticas\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 11.40}Um DBA possui acesso para alterar dados de negócio e também apagar logs de auditoria. O principal problema é:\begin{enumerate}[label=\Alph*)]\item segregação de funções e confiabilidade da trilha de auditoria\item normalização insuficiente\item ausência de ETL\item falta de data lake\item cardinalidade baixa\end{enumerate}\end{questao}

\subsection*{N4 — Avançado / Auditoria}

\begin{questao}{\nivel{N4} 11.41}Um pipeline de fiscalização recebe eventos com reenvio após falhas e grava cada pagamento sem chave idempotente, inflando valores. A recomendação principal é:\begin{enumerate}[label=\Alph*)]\item usar identificador estável, deduplicação/transação e processamento idempotente\item desabilitar todo retry\item remover chave única\item trocar relacional por NoSQL sem mudar lógica\item aumentar partições apenas\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 11.42}Um órgão publica dataset aberto extraído do sistema transacional, mas sem catálogo, definição de campos, periodicidade ou versão. Qual risco é mais relevante?\begin{enumerate}[label=\Alph*)]\item baixa interpretabilidade, reprodutibilidade e governança do dado publicado\item excesso de normalização\item impossibilidade de usar HTTP\item falta de RAID\item ausência de VLAN\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 11.43}Uma carga ETL apaga a tabela destino e recarrega 2 bilhões de linhas diariamente. O SLA está sendo violado. Qual abordagem deve ser avaliada?\begin{enumerate}[label=\Alph*)]\item carga incremental/CDC, particionamento e estratégia de atualização compatível com a semântica\item aumentar full scan\item eliminar índices permanentemente sem análise\item converter tudo para planilha\item remover validações\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 11.44}Auditoria identifica que relatório financeiro usa dimensão de fornecedor atualizada por sobrescrita, apagando razão social histórica e reescrevendo análises antigas. O problema é:\begin{enumerate}[label=\Alph*)]\item gestão inadequada de histórico dimensional para requisito temporal\item falta de NAT\item conflito de deadlock\item ausência de RAID 10\item erro de TCP\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 11.45}Uma base de risco foi construída juntando dados por nome de pessoa, causando falsos matches entre homônimos. Qual melhoria é prioritária?\begin{enumerate}[label=\Alph*)]\item adotar regras de entity resolution com identificadores confiáveis, score e validação de ambiguidade\item aumentar tamanho da fonte do relatório\item usar apenas LIKE em nome\item remover metadados\item ignorar duplicidades\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 11.46}O data lake permite que qualquer analista leia dados pessoais brutos ``porque o ambiente é interno''. Qual princípio de controle foi violado?\begin{enumerate}[label=\Alph*)]\item mínimo privilégio e governança de acesso/classificação\item integridade referencial apenas\item normalização 2FN\item compressão\item particionamento\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 11.47}Um dashboard soma valores de contrato após JOIN 1:N com pagamentos e apresenta total duplicado. Qual procedimento de auditoria é mais adequado?\begin{enumerate}[label=\Alph*)]\item verificar granularidade de cada fonte, cardinalidade do JOIN e reconciliar total com fonte independente\item confiar no dashboard porque há SQL\item adicionar DISTINCT sem entender semântica\item apagar pagamentos duplicados automaticamente\item aumentar CPU\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 11.48}Uma equipe mantém backup de banco, mas nunca testa point-in-time recovery. O banco usa WAL/binlog para recuperação. Qual conclusão é correta?\begin{enumerate}[label=\Alph*)]\item existência dos logs não prova recuperabilidade; cadeia, retenção e restauração precisam ser testadas\item WAL substitui full backup em qualquer caso\item PITR independe de logs\item backup não precisa de consistência\item teste é desnecessário se storage é RAID\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 11.49}Um SGBD foi particionado por data, mas quase todas as consultas filtram por órgão e varrem todas as partições. Qual avaliação é correta?\begin{enumerate}[label=\Alph*)]\item estratégia de particionamento pode estar desalinhada ao padrão de acesso e deve ser reavaliada com plano/métricas\item particionamento sempre melhora qualquer consulta\item basta adicionar mais partições\item o problema é normalização\item partições substituem índices\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 11.50}Um pipeline de dados críticos não registra versão do código de transformação nem parâmetros. O relatório muda e ninguém consegue explicar por quê. O controle faltante é principalmente:\begin{enumerate}[label=\Alph*)]\item versionamento, reprodutibilidade e linhagem de transformação\item DHCP\item VLAN\item RAID\item autenticação Wi-Fi\end{enumerate}\end{questao}
