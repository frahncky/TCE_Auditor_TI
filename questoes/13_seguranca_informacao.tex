\section{Banco de questões — Segurança da Informação}

\subsection*{N1 — Fundamento competitivo}

\begin{questao}{\nivel{N1} 13.01 — Cebraspe/Certo ou Errado}
Duas credenciais distintas baseadas em conhecimento, como senha e PIN, não constituem autenticação multifator apenas por serem dois segredos diferentes.
\end{questao}

\begin{questao}{\nivel{N1} 13.02 — Cebraspe/Certo ou Errado}
Assinatura digital, por si só, garante confidencialidade do conteúdo assinado, pois utiliza criptografia assimétrica.
\end{questao}

\begin{questao}{\nivel{N1} 13.03 — Cebraspe/Certo ou Errado}
A criptografia de disco reduz risco de exposição em caso de perda física da mídia, mas pode ser insuficiente contra atacante que compromete o sistema em execução e obtém acesso às chaves.
\end{questao}

\begin{questao}{\nivel{N1} 13.04 — Cebraspe/Certo ou Errado}
Um IDS e um IPS podem empregar técnicas de detecção semelhantes, mas diferem porque o IPS pode atuar inline e bloquear tráfego segundo política.
\end{questao}

\begin{questao}{\nivel{N1} 13.05 — Cebraspe/Certo ou Errado}
O uso de VPN garante que o dispositivo remoto esteja íntegro, pois o túnel criptografado impede que malware já presente no endpoint acesse a rede corporativa.
\end{questao}

\begin{questao}{\nivel{N1} 13.06 — Cebraspe/Certo ou Errado}
Hash criptográfico preservado desde a coleta de um arquivo pode demonstrar que a cópia não foi alterada, mas não prova que a origem do arquivo era legítima ou que o conteúdo era verdadeiro.
\end{questao}

\begin{questao}{\nivel{N1} 13.07 — Cebraspe/Certo ou Errado}
A certificação ISO/IEC 27001 de um provedor de nuvem implica, automaticamente, que qualquer workload implantado pelo cliente nesse provedor está coberto pelo mesmo escopo de certificação.
\end{questao}

\begin{questao}{\nivel{N1} 13.08 — Cebraspe/Certo ou Errado}
Risco residual é o risco que permanece após a aplicação dos controles, enquanto risco inerente é avaliado antes de considerar esses controles.
\end{questao}

\begin{questao}{\nivel{N1} 13.09 — Cebraspe/Certo ou Errado}
Uma DMZ bem projetada reduz exposição da rede interna, mas não deve ser interpretada como zona sem controles de segurança.
\end{questao}

\begin{questao}{\nivel{N1} 13.10 — Cebraspe/Certo ou Errado}
A existência de backup, sem teste de restauração, é evidência suficiente de que o RTO definido no plano de continuidade será cumprido.
\end{questao}

\subsection*{N2 — Aplicação}

\begin{questao}{\nivel{N2} 13.11 — FGV/IAM}
Um administrador usa a mesma conta para e-mail, navegação e administração de controladores de domínio. A principal recomendação é:
\begin{enumerate}[label=\Alph*)]
\item manter a conta única para simplificar auditoria;
\item separar identidade administrativa da conta cotidiana e restringir seu uso;
\item compartilhar a conta com a equipe;
\item remover MFA para reduzir atrito;
\item manter a conta, mas trocar a senha semanalmente.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N2} 13.12 — FCC/MFA}
Usuários recebem dezenas de notificações push de MFA até que alguns aprovam por engano. A técnica explorada é conhecida como:
\begin{enumerate}[label=\Alph*)]
\item SQL injection;
\item MFA fatigue/push bombing;
\item DNS poisoning;
\item session fixation apenas;
\item ARP spoofing.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N2} 13.13 — Cebraspe/Certo ou Errado}
Mecanismos resistentes a phishing, number matching e limitação de prompts podem reduzir ataques de MFA fatigue, enquanto simplesmente aumentar o número de notificações tende a piorar o problema.
\end{questao}

\begin{questao}{\nivel{N2} 13.14 — FGV/web security}
Uma aplicação concatena diretamente entrada do usuário em consulta SQL. O controle mais importante é:
\begin{enumerate}[label=\Alph*)]
\item prepared statements/consultas parametrizadas, aliados a privilégio mínimo;
\item apenas WAF;
\item criptografia de disco;
\item RAID 6;
\item aumentar o timeout do banco.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N2} 13.15 — Cebraspe/Certo ou Errado}
Um WAF pode atuar como controle compensatório contra SQL injection, mas não remove a vulnerabilidade do código e pode ser contornado.
\end{questao}

\begin{questao}{\nivel{N2} 13.16 — FCC/PKI}
Um serviço fica indisponível porque o certificado TLS expirou. O incidente evidencia falha de:
\begin{enumerate}[label=\Alph*)]
\item gestão do ciclo de vida de certificados;
\item integridade referencial;
\item roteamento BGP;
\item política de RAID;
\item normalização de banco.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N2} 13.17 — FGV/ransomware}
Backups online usam as mesmas credenciais de domínio da produção. Após comprometimento de administrador, o atacante apaga backups antes de cifrar os servidores. O controle mais adequado é:
\begin{enumerate}[label=\Alph*)]
\item aumentar a capacidade do storage;
\item separar credenciais, adotar cópias imutáveis/offline e testar restauração;
\item manter os backups montados permanentemente;
\item usar RAID 0;
\item desabilitar logs.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N2} 13.18 — Cebraspe/Certo ou Errado}
Um backup imutável reduz risco de destruição pelo atacante, mas não elimina a necessidade de testar integridade, retenção e processo de recuperação.
\end{questao}

\begin{questao}{\nivel{N2} 13.19 — FCC/continuidade}
Um restore test recupera dados de 10 minutos antes da falha e leva 4 horas. Os objetivos eram RPO de 30 minutos e RTO de 2 horas. Nesse teste:
\begin{enumerate}[label=\Alph*)]
\item ambos foram cumpridos;
\item somente o RPO foi cumprido;
\item somente o RTO foi cumprido;
\item nenhum foi cumprido;
\item não é possível avaliar.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N2} 13.20 — FGV/segurança de rede}
Uma VPN de acesso remoto entrega conectividade ampla a toda a rede após autenticação, sem avaliar postura do dispositivo. O principal risco é:
\begin{enumerate}[label=\Alph*)]
\item endpoint comprometido obter movimento lateral amplo;
\item perda automática de confidencialidade do túnel;
\item impossibilidade de MFA;
\item falta de DNSSEC;
\item perda de integridade do certificado por definição.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N2} 13.21 — Cebraspe/Certo ou Errado}
Privilégio mínimo e segmentação continuam necessários mesmo quando a conexão utiliza VPN criptografada.
\end{questao}

\begin{questao}{\nivel{N2} 13.22 — FCC/criptografia}
Uma aplicação precisa proteger grandes volumes de dados após estabelecer identidade das partes. O desenho mais comum é:
\begin{enumerate}[label=\Alph*)]
\item usar apenas criptografia assimétrica para todo o volume;
\item usar arquitetura híbrida, com assimétrica para autenticação/acordo e simétrica para os dados;
\item usar hash como cifra reversível;
\item usar assinatura digital para confidencialidade;
\item usar certificado sem chave de sessão.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N2} 13.23 — Cebraspe/Certo ou Errado}
Reutilização incorreta de nonce em determinados modos autenticados de criptografia pode comprometer a segurança mesmo quando o algoritmo de cifra subjacente é robusto.
\end{questao}

\begin{questao}{\nivel{N2} 13.24 — FGV/gestão de risco}
Um risco tem perda estimada de R\$ 400 mil por evento e frequência média de 0,25 evento/ano. A perda anual esperada simplificada é:
\begin{enumerate}[label=\Alph*)]
\item R\$ 25 mil;
\item R\$ 50 mil;
\item R\$ 100 mil;
\item R\$ 400 mil;
\item R\$ 1,6 milhão.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N2} 13.25 — Cebraspe/Certo ou Errado}
Aceitar um risco significa decisão consciente e autorizada sobre o risco residual, e não simples ausência de tratamento documentado.
\end{questao}

\subsection*{N3 — Integração de concurso}

\begin{questao}{\nivel{N3} 13.26 — FGV/assertivas}
Considere:

I. ISO/IEC 27001 estabelece requisitos para um SGSI.

II. ISO/IEC 27002 fornece orientação sobre controles.

III. ISO/IEC 27005 trata gestão de riscos de segurança da informação.

Assinale a alternativa correta.
\begin{enumerate}[label=\Alph*)]
\item Apenas I.
\item Apenas I e II.
\item Apenas II e III.
\item I, II e III.
\item Apenas III.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N3} 13.27 — Cebraspe/Certo ou Errado}
A certificação ISO/IEC 27001 deve ser interpretada segundo seu escopo; um incidente fora desse escopo não demonstra, por si só, que a certificação era inválida.
\end{questao}

\begin{questao}{\nivel{N3} 13.28 — FCC/forense}
Durante coleta de evidência, uma imagem de disco é copiada, hashada e armazenada. Porém, o analista não registrou quem teve acesso ao equipamento antes da coleta. O principal problema remanescente é:
\begin{enumerate}[label=\Alph*)]
\item o hash não serve para integridade;
\item a cadeia de custódia/origem pode estar incompleta, apesar da integridade da cópia posterior;
\item o arquivo deixa de ser digital;
\item assinatura digital torna-se obrigatoriamente inválida;
\item o hash prova autoria.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N3} 13.29 — FGV/logs}
Um SIEM recebe logs de todos os servidores, mas os relógios diferem até 25 minutos. O impacto principal é:
\begin{enumerate}[label=\Alph*)]
\item perda de confidencialidade dos logs;
\item dificuldade de correlação temporal e reconstrução causal;
\item impossibilidade de usar TLS;
\item redução de espaço;
\item aumento de disponibilidade.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N3} 13.30 — Cebraspe/Certo ou Errado}
Uma conta compartilhada entre administradores pode continuar autenticando corretamente, mas fragiliza accountability porque os logs deixam de identificar de forma confiável o indivíduo que executou a ação.
\end{questao}

\begin{questao}{\nivel{N3} 13.31 — FCC/ataque distribuído}
Milhares de IPs realizam poucas tentativas de uma senha comum contra muitas contas. O padrão é mais compatível com:
\begin{enumerate}[label=\Alph*)]
\item password spraying distribuído;
\item buffer overflow;
\item SQL injection;
\item ARP poisoning;
\item ransomware de dupla extorsão.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N3} 13.32 — FGV/DMZ}
Um servidor web em DMZ possui conexão administrativa irrestrita para toda a rede interna. O risco principal é:
\begin{enumerate}[label=\Alph*)]
\item a DMZ deixa de ser rede TCP/IP;
\item comprometimento do servidor exposto servir como pivô para a rede interna;
\item impossibilidade de usar TLS;
\item perda automática de certificado;
\item aumento de RPO.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N3} 13.33 — Cebraspe/Certo ou Errado}
A criptografia de dados em repouso e a criptografia do canal de rede protegem ameaças distintas e podem ser necessárias simultaneamente.
\end{questao}

\begin{questao}{\nivel{N3} 13.34 — FCC/segredo em aplicação}
A aplicação armazena segredo de API no repositório e o segredo é usado também por produção. Mesmo após remover o arquivo do branch atual, a ação prioritária é:
\begin{enumerate}[label=\Alph*)]
\item manter a chave e reescrever o histórico somente;
\item rotacionar/revogar o segredo e migrar para mecanismo apropriado de gestão;
\item alterar o nome da variável;
\item aumentar o timeout;
\item apenas tornar o repositório privado.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N3} 13.35 — FGV/BIA}
Um processo crítico gera impacto financeiro baixo nas primeiras duas horas, mas após oito horas paralisa serviço essencial ao cidadão. A BIA deve:
\begin{enumerate}[label=\Alph*)]
\item considerar apenas impacto financeiro imediato;
\item considerar evolução do impacto ao longo do tempo e dependências do processo;
\item ignorar serviços públicos sem receita;
\item usar somente RPO;
\item ser substituída por pentest.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N3} 13.36 — Cebraspe/Certo ou Errado}
RTO e RPO tratam dimensões distintas: tempo para restaurar serviço e perda temporal tolerável de dados, respectivamente.
\end{questao}

\begin{questao}{\nivel{N3} 13.37 — FCC/ransomware}
Após ransomware, a equipe deseja restaurar imediatamente os servidores sem investigar vetor de entrada ou credenciais comprometidas. O principal risco é:
\begin{enumerate}[label=\Alph*)]
\item recuperação lenta;
\item reinfecção/recorrência ao restaurar ambiente sem remover a causa e conter o atacante;
\item perda de RAID;
\item excesso de logs;
\item aumento de confidencialidade.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N3} 13.38 — Cebraspe/Certo ou Errado}
Pagamento de resgate não garante recuperação dos dados nem elimina risco de publicação de informações previamente exfiltradas.
\end{questao}

\begin{questao}{\nivel{N3} 13.39 — FGV/honeypot}
Um honeypot de alta interação possui saída irrestrita para Internet e rede interna. A principal deficiência é:
\begin{enumerate}[label=\Alph*)]
\item o honeypot não pode ser monitorado;
\item o ativo comprometido pode ser usado como pivô; egress e isolamento devem ser controlados;
\item honeypots nunca devem usar rede;
\item a solução é eliminar logging;
\item o honeypot deve usar conta Domain Admin.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N3} 13.40 — Cebraspe/Certo ou Errado}
Uma política de senha que exige trocas muito frequentes pode induzir padrões previsíveis; a eficácia deve ser analisada junto com MFA, detecção de comprometimento e risco, e não apenas pela periodicidade.
\end{questao}

\subsection*{N4 — Auditoria / avançado}

\begin{questao}{\nivel{N4} 13.41 — Cebraspe/caso integrado}
Administradores podem alterar dados de negócio e apagar seus próprios logs. Julgue o item:

Ainda que todas as ações exijam autenticação, permanece fragilidade relevante de segregação de funções e integridade da trilha de auditoria.
\end{questao}

\begin{questao}{\nivel{N4} 13.42 — FGV/ransomware}
Um órgão possui backup diário, mas todas as cópias ficam online no mesmo domínio e não há restore test há dois anos. O achado mais adequado é:
\begin{enumerate}[label=\Alph*)]
\item ``há backup, logo o risco é baixo'';
\item ``a estratégia não demonstra resiliência a comprometimento do domínio nem recuperabilidade; faltam segregação/imutabilidade e testes'';
\item ``o problema é somente frequência'';
\item ``RAID resolve a deficiência'';
\item ``basta aumentar a retenção''.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N4} 13.43 — FCC/Zero Trust}
Uma organização considera qualquer dispositivo conectado à LAN corporativa como confiável e concede acesso amplo após autenticação inicial. A abordagem mais coerente com Zero Trust é:
\begin{enumerate}[label=\Alph*)]
\item manter confiança por localização;
\item verificar identidade, contexto e postura continuamente, com privilégio mínimo e segmentação;
\item remover MFA;
\item conceder administrador para reduzir falhas;
\item usar apenas firewall perimetral.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N4} 13.44 — Cebraspe/Certo ou Errado}
Uma VPN criptografada entre filiais pode proteger confidencialidade do tráfego e, ao mesmo tempo, transportar malware legitimamente enviado por uma estação comprometida.
\end{questao}

\begin{questao}{\nivel{N4} 13.45 — FGV/risco}
A matriz de risco classifica ransomware como ``baixo impacto'' porque considera apenas custo de reposição de hardware. A falha metodológica é:
\begin{enumerate}[label=\Alph*)]
\item ignorar efeitos de indisponibilidade, dados, sanções, missão e reputação;
\item usar impacto em vez de probabilidade;
\item considerar hardware;
\item não usar hash;
\item ausência de DNSSEC.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N4} 13.46 — Cebraspe/Certo ou Errado}
Um certificado de conformidade deve ser avaliado em conjunto com seu escopo, período e controles efetivamente operados; sua existência isolada não substitui procedimentos de auditoria.
\end{questao}

\begin{questao}{\nivel{N4} 13.47 — FCC/forense}
Durante incidente, logs são exportados para arquivo e o hash é calculado. Depois, descobre-se que o servidor de origem permitia ao administrador alterar logs antes da exportação. O hash prova:
\begin{enumerate}[label=\Alph*)]
\item que os logs originais eram verdadeiros;
\item apenas que o arquivo exportado não mudou após o hash, não que o conteúdo anterior era íntegro;
\item autoria dos eventos;
\item ausência de comprometimento;
\item confidencialidade dos logs.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N4} 13.48 — FGV/resposta a incidentes}
Após suspeita de exfiltração, a equipe apaga imediatamente os discos comprometidos para ``limpar o ambiente''. O principal problema é:
\begin{enumerate}[label=\Alph*)]
\item limpeza rápida sempre melhora investigação;
\item destruição de evidência antes de compreender escopo, vetor e cadeia de ataque;
\item excesso de retenção;
\item falta de RAID;
\item impossibilidade de usar TLS.
\end{enumerate}
\end{questao}

\begin{questao}{\nivel{N4} 13.49 — Cebraspe/Certo ou Errado}
Em continuidade, restaurar arquivos com sucesso não é suficiente para demonstrar recuperação do serviço se banco, integrações, identidades e procedimentos de retorno ao tráfego não forem testados.
\end{questao}

\begin{questao}{\nivel{N4} 13.50 — FGV/matriz de achado}
Em revisão de IAM, 38 contas de ex-servidores permanecem ativas, 9 com privilégios elevados. Qual achado está mais bem formulado?
\begin{enumerate}[label=\Alph*)]
\item ``O Active Directory é inseguro'';
\item ``O RH falhou, logo toda a TI é irregular'';
\item ``A ausência de integração eficaz entre desligamento e revogação de acesso mantém contas sem necessidade e amplia risco de uso indevido; recomenda-se automatizar offboarding, recertificar acessos e tratar privilégios'';
\item ``Basta trocar senhas'';
\item ``Contas inativas não representam risco''.
\end{enumerate}
\end{questao}
