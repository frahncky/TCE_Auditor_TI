\section{Banco de questões — Gestão e Governança de TI}

\subsection*{N1 — Fundamentos}
\begin{questao}{\nivel{N1} 14.01}No COBIT 2019, o domínio de governança é:\begin{enumerate}[label=\Alph*)]\item EDM\item APO\item BAI\item DSS\item MEA\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 14.02}Governança distingue-se de gestão porque a governança enfatiza:\begin{enumerate}[label=\Alph*)]\item avaliar, direcionar e monitorar\item executar tickets\item configurar servidores\item programar aplicações\item administrar banco diariamente\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 14.03}No COBIT, APO significa:\begin{enumerate}[label=\Alph*)]\item Align, Plan and Organize\item Assess, Protect and Operate\item Audit, Plan and Optimize\item Align, Produce and Outsource\item Apply, Perform and Observe\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 14.04}ITIL 4 organiza a criação de valor por meio do:\begin{enumerate}[label=\Alph*)]\item Service Value System\item RAID\item TCP/IP\item PMO apenas\item data warehouse\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 14.05}Em ITIL 4, incident management busca:\begin{enumerate}[label=\Alph*)]\item restaurar operação normal rapidamente e reduzir impacto\item encontrar sempre a causa raiz antes de restaurar\item gerenciar portfólio de projetos\item criar orçamento\item administrar contratos apenas\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 14.06}Problem management busca principalmente:\begin{enumerate}[label=\Alph*)]\item reduzir probabilidade/impacto de incidentes por tratamento de causas e erros conhecidos\item atender requisição de senha apenas\item substituir change enablement\item restaurar serviço sem investigar\item gerenciar fornecedores exclusivamente\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 14.07}PETI relaciona-se mais diretamente a:\begin{enumerate}[label=\Alph*)]\item direcionamento estratégico de TI\item backup diário\item código-fonte\item firewall\item teste unitário\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 14.08}KPI é:\begin{enumerate}[label=\Alph*)]\item indicador-chave de desempenho\item protocolo de criptografia\item tipo de banco\item padrão de rede\item linguagem de programação\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 14.09}Em BPMN, gateway exclusivo representa:\begin{enumerate}[label=\Alph*)]\item escolha de um caminho conforme condição\item paralelismo obrigatório de todos os caminhos\item início do processo apenas\item mensagem entre pools\item banco de dados\end{enumerate}\end{questao}
\begin{questao}{\nivel{N1} 14.10}DAMA-DMBOK trata principalmente de:\begin{enumerate}[label=\Alph*)]\item gestão e governança de dados\item roteamento\item virtualização\item criptografia\item gestão de incidentes apenas\end{enumerate}\end{questao}

\subsection*{N2 — Aplicação}
\begin{questao}{\nivel{N2} 14.11}Qual situação representa governança?\begin{enumerate}[label=\Alph*)]\item conselho avalia opções de sourcing, direciona escolha e monitora valor/risco\item técnico instala patch\item DBA restaura backup\item analista fecha chamado\item desenvolvedor corrige bug\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.12}No COBIT, BAI está associado a:\begin{enumerate}[label=\Alph*)]\item construir, adquirir e implementar\item apenas governança\item entregar suporte exclusivamente\item monitorar auditoria apenas\item recursos humanos apenas\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.13}MEA no COBIT relaciona-se a:\begin{enumerate}[label=\Alph*)]\item monitorar, avaliar e analisar\item desenvolver software\item operar service desk apenas\item planejar projetos apenas\item adquirir hardware\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.14}Um incidente é restaurado, mas reaparece toda semana. Qual prática ITIL deve atuar sobre causa?\begin{enumerate}[label=\Alph*)]\item Problem management\item Incident management somente\item Service request management\item Relationship management\item Deployment management\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.15}Uma solicitação padrão de criação de usuário é mais compatível com:\begin{enumerate}[label=\Alph*)]\item service request management\item problem management\item portfolio management\item incident maior necessariamente\item risk management exclusivamente\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.16}Em gestão de risco ISO 31000, identificação, análise e avaliação compõem:\begin{enumerate}[label=\Alph*)]\item avaliação de riscos\item tratamento apenas\item auditoria financeira\item controle de mudança\item service desk\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.17}Apetite a risco é:\begin{enumerate}[label=\Alph*)]\item quantidade/tipo de risco que organização está disposta a buscar ou reter em busca de objetivos\item qualquer risco residual\item custo do incidente\item vulnerabilidade técnica\item probabilidade zero\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.18}Um PDTI sem responsáveis e metas possui deficiência porque:\begin{enumerate}[label=\Alph*)]\item não transforma estratégia em execução monitorável\item contém governança demais\item elimina riscos\item substitui orçamento\item torna KPI desnecessário\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.19}Um KPI deve, idealmente, ter:\begin{enumerate}[label=\Alph*)]\item definição, fórmula, fonte, periodicidade, responsável e meta/interpretação\item somente nome\item apenas cor no dashboard\item valor sem unidade\item nenhuma fonte\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.20}BSC busca:\begin{enumerate}[label=\Alph*)]\item traduzir estratégia em objetivos e indicadores relacionados em perspectivas\item substituir toda gestão de risco\item ser somente painel financeiro\item medir CPU\item criar banco\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.21}Em BPMN, fluxo de sequência:\begin{enumerate}[label=\Alph*)]\item conecta elementos dentro do mesmo processo/pool e não deve atravessar pools\item atravessa pools para mensagens\item substitui gateway\item representa banco\item é somente anotação\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.22}Fluxo de mensagem em BPMN representa:\begin{enumerate}[label=\Alph*)]\item comunicação entre participantes/pools\item sequência interna obrigatória\item associação de dados apenas\item gateway paralelo\item evento de timer\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.23}Compliance significa:\begin{enumerate}[label=\Alph*)]\item aderência a obrigações legais, regulatórias, contratuais e internas aplicáveis\item ausência de risco\item eficiência máxima\item segurança absoluta\item inovação\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.24}Governança de dados deve definir:\begin{enumerate}[label=\Alph*)]\item autoridade, ownership, políticas, qualidade e decisões sobre dados\item somente tamanho do banco\item apenas tecnologia ETL\item apenas backup\item somente Python\end{enumerate}\end{questao}
\begin{questao}{\nivel{N2} 14.25}LAI e LGPD devem ser aplicadas de forma:\begin{enumerate}[label=\Alph*)]\item compatível, equilibrando transparência com proteção e bases legais/restrições\item excludente: LGPD sempre impede LAI\item excludente: LAI sempre vence LGPD\item sem classificar dados\item apenas a dados privados\end{enumerate}\end{questao}

\subsection*{N3 — Nível de concurso}
\begin{questao}{\nivel{N3} 14.26}No COBIT 2019, ``monitorar'' pode aparecer em governança e gestão. Como diferenciar?\begin{enumerate}[label=\Alph*)]\item pelo objeto, nível de decisão e responsabilidade, não pela palavra isolada\item toda atividade de monitorar é EDM\item toda atividade de monitorar é MEA\item não existe monitoramento em gestão\item não existe monitoramento em governança\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.27}A cascata de objetivos do COBIT busca:\begin{enumerate}[label=\Alph*)]\item traduzir necessidades de stakeholders em objetivos empresariais, de alinhamento e de governança/gestão\item gerar tickets\item criar RAID\item substituir estratégia\item eliminar tailoring\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.28}Tailoring em governança significa:\begin{enumerate}[label=\Alph*)]\item adaptar sistema de governança ao contexto, estratégia, risco e fatores de desenho\item aplicar framework idêntico a todos\item remover controles\item escolher somente tecnologia\item ignorar objetivos\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.29}No ITIL 4, a cadeia de valor inclui:\begin{enumerate}[label=\Alph*)]\item Plan, Improve, Engage, Design \& Transition, Obtain/Build, Deliver \& Support\item apenas Incident e Problem\item somente projetos\item somente operações\item apenas fornecedores\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.30}``Start where you are'' orienta a organização a:\begin{enumerate}[label=\Alph*)]\item avaliar estado atual e aproveitar o que funciona antes de reconstruir tudo\item nunca mudar\item copiar benchmark sem análise\item ignorar dados atuais\item eliminar automação\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.31}Change enablement busca:\begin{enumerate}[label=\Alph*)]\item maximizar mudanças bem-sucedidas avaliando risco, autorização e agenda adequada\item impedir toda mudança\item aprovar tudo automaticamente\item substituir deploy\item tratar só incidentes\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.32}Um PMO exige mesmo processo preditivo de 50 documentos para qualquer iniciativa, inclusive experimentos de 2 semanas. Qual princípio é violado?\begin{enumerate}[label=\Alph*)]\item tailoring/proporcionalidade ao contexto\item governança\item documentação\item compliance\item monitoramento\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.33}No PMBOK 8, uma visão madura de projetos deve enfatizar:\begin{enumerate}[label=\Alph*)]\item valor, princípios, domínios/processos adaptados e resultados\item lista rígida universal de 49 processos da 6ª edição\item cronograma como único sucesso\item exclusão de métodos ágeis\item ausência de governança\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.34}Um indicador ``número de chamados fechados'' causa encerramentos prematuros. Isso exemplifica:\begin{enumerate}[label=\Alph*)]\item comportamento disfuncional induzido por métrica mal desenhada\item indicador perfeito\item aumento automático de valor\item ausência de Goodhart\item conformidade\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.35}Para reduzir gaming, um conjunto de indicadores deve:\begin{enumerate}[label=\Alph*)]\item equilibrar quantidade, qualidade, tempo, resultado e efeitos adversos\item usar apenas volume\item ocultar fórmula\item não ter owner\item não ter fonte\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.36}Gateway paralelo em BPMN é usado para:\begin{enumerate}[label=\Alph*)]\item dividir/sincronizar caminhos concorrentes sem condição exclusiva\item escolher um único caminho\item enviar mensagem entre pools apenas\item criar evento\item encerrar processo obrigatoriamente\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.37}Um processo modelado com sequência atravessando dois pools está:\begin{enumerate}[label=\Alph*)]\item semanticamente inadequado; comunicação entre pools usa fluxo de mensagem\item correto em qualquer caso\item mais seguro\item obrigatório no BPMN\item equivalente a gateway\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.38}Conformidade com norma não garante segurança porque:\begin{enumerate}[label=\Alph*)]\item escopo, implementação, risco residual e ameaças podem exigir controles além do mínimo\item compliance e segurança são idênticos\item norma elimina risco\item auditoria é desnecessária\item certificação cobre qualquer sistema\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.39}Um dado pessoal é publicado por transparência ativa sem análise de necessidade ou base legal. O problema é:\begin{enumerate}[label=\Alph*)]\item ausência de balanceamento LAI/LGPD e governança de divulgação\item transparência sempre autoriza tudo\item LGPD nunca se aplica ao poder público\item dado público nunca é pessoal\item anonimização é desnecessária\end{enumerate}\end{questao}
\begin{questao}{\nivel{N3} 14.40}Governança de dados sem definição de data owner tende a gerar:\begin{enumerate}[label=\Alph*)]\item responsabilidade difusa sobre qualidade, acesso e significado\item maior accountability\item eliminação de conflitos\item normalização automática\item segurança garantida\end{enumerate}\end{questao}

\subsection*{N4 — Avançado / Auditoria}
\begin{questao}{\nivel{N4} 14.41}Um TCE mede TI apenas por projetos entregues ``no prazo'', embora haja baixa adoção e alto custo operacional. A recomendação é:\begin{enumerate}[label=\Alph*)]\item usar métricas ligadas a valor, benefícios, adoção, qualidade, risco e custo\item manter só prazo\item medir número de servidores\item eliminar metas\item trocar todos projetos por Scrum\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 14.42}O PDTI possui 80 iniciativas sem vínculo com objetivos institucionais. O risco é:\begin{enumerate}[label=\Alph*)]\item portfólio orientado por demanda local sem demonstrar alinhamento e valor estratégico\item excesso de governança\item pouca tecnologia\item ausência de Scrum\item falta de firewall\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 14.43}O comitê de TI aprova todas as contratações, mas não acompanha benefícios após implantação. Qual lacuna?\begin{enumerate}[label=\Alph*)]\item governança incompleta, pois direção sem monitoramento de valor/resultado não fecha ciclo\item operação excessiva\item ausência de RAID\item falta de código\item excesso de indicadores\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 14.44}Auditoria encontra 40 KPIs, mas nenhum possui fonte ou responsável. A conclusão é:\begin{enumerate}[label=\Alph*)]\item medição não é suficientemente governada/reproduzível para accountability\item mais KPIs sempre é melhor\item basta dashboard bonito\item fonte é desnecessária\item owner é incompatível com BSC\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 14.45}Uma meta de disponibilidade é atingida porque janelas de indisponibilidade foram excluídas manualmente sem regra. O problema é:\begin{enumerate}[label=\Alph*)]\item indicador manipulável e sem definição operacional consistente\item disponibilidade é impossível de medir\item exclusão manual sempre é válida\item SLO não precisa de fórmula\item SLA dispensa evidência\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 14.46}Uma organização implementa COBIT copiando todos os objetivos sem priorização. Isso contraria:\begin{enumerate}[label=\Alph*)]\item fatores de desenho e tailoring orientado a objetivos/risco\item governança\item documentação\item capacidade\item auditoria\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 14.47}A unidade de TI é responsável por definir seu próprio apetite a risco sem participação da alta administração. A avaliação é:\begin{enumerate}[label=\Alph*)]\item inadequada; apetite deve refletir objetivos e responsabilidade de governança institucional\item correta por ser tema técnico\item risco é responsabilidade só de TI\item alta administração não participa de governança\item apetite é igual a vulnerabilidade\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 14.48}Um processo BPMN possui cinco aprovações sequenciais que não agregam decisão distinta. A abordagem de melhoria deve:\begin{enumerate}[label=\Alph*)]\item analisar valor, risco e redundância, simplificando controles sem perder accountability\item manter todas por tradição\item automatizar desperdício sem questionar\item adicionar mais aprovações\item eliminar qualquer controle\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 14.49}Em governança de dados, duas áreas possuem definições diferentes de ``contrato ativo''. Qual controle é prioritário?\begin{enumerate}[label=\Alph*)]\item glossário/definição oficial com owner e processo de decisão sobre semântica\item aumentar armazenamento\item duplicar dashboards\item criar mais bancos\item remover metadados\end{enumerate}\end{questao}
\begin{questao}{\nivel{N4} 14.50}Uma organização certificada em ISO 27001 sofre incidente causado por configuração fora do escopo certificado. Qual conclusão?\begin{enumerate}[label=\Alph*)]\item certificação não elimina risco e deve ser interpretada conforme escopo e operação real\item certificação prova impossibilidade de incidente\item incidente invalida toda ISO\item escopo é irrelevante\item auditoria não deve considerar configuração\end{enumerate}\end{questao}
